SELECT
  raw.Brand,
  raw.LOB,
  raw.Channel_Name,
  COUNT(*) AS row_count,
  COUNT(DISTINCT raw.Date) AS days_present,
  MIN(raw.Date) AS min_date,
  MAX(raw.Date) AS max_date,
  SUM(raw.Spend) AS spend

FROM prdrzranalytics.lab42.media_analytics_integrated_snapshot raw

WHERE raw.Channel_Group_Name = 'Paid Social'
  AND raw.Date >= date_sub(current_date(), 35)

GROUP BY
  raw.Brand,
  raw.LOB,
  raw.Channel_Name

ORDER BY
  spend DESC;
