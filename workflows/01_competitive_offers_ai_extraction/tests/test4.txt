SELECT
  TRY_CAST(Week_Beginning_Monday AS DATE) AS Week_Beginning_Monday,
  SUM(Spend) AS total_spend
FROM prdrzranalytics.lab42.raw_media_flowchart
WHERE TRY_CAST(CAST(FileLoad_Date AS STRING) AS DATE) = '2026-08-03'
  AND UPPER(TRIM(LOB_Supported)) = 'CONSUMER POSTPAID'
  AND Channel IS NOT NULL
  AND Channel NOT IN ('OTHER (do not use)', 'Non-Working', 'Budget Held')
  AND TRY_CAST(Week_Beginning_Monday AS DATE) IN ('2026-06-29','2026-07-06','2026-07-13','2026-07-20','2026-07-27')
GROUP BY TRY_CAST(Week_Beginning_Monday AS DATE)
ORDER BY Week_Beginning_Monday;

output:
Week_Beginning_Monday	total_spend
2026-06-29	13540894.864449767
2026-07-06	19866424.311832428
2026-07-13	24248646.863773346
2026-07-20	19783145.137821198
2026-07-27	21789466.85049057

SELECT
  Week_Beginning_Monday,
  SUM(COALESCE(Spend_Actual, Spend_Forecast)) AS total_spend
FROM prdrzranalytics.lab42.sdi_vw_mfc_gold_spend_weekly
WHERE LOB_Supported = 'CONSUMER POSTPAID'
  AND Week_Beginning_Monday IN ('2026-06-29','2026-07-06','2026-07-13','2026-07-20','2026-07-27')
GROUP BY Week_Beginning_Monday
ORDER BY Week_Beginning_Monday;
