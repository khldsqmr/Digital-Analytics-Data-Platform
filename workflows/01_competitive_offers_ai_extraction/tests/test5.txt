SELECT date, campaign_id, campaign_name, spend
FROM prd_dbi_analytics.improvado.mrt_paidsocial_pivot
WHERE lob = 'PrePaid' AND channel = 'Pinterest' AND date BETWEEN '2026-08-01' AND '2026-08-16'
ORDER BY date;


SELECT channel, lob, COUNT(*) AS n_rows, SUM(CASE WHEN spend IS NULL THEN 1 ELSE 0 END) AS n_null_spend
FROM prd_dbi_analytics.improvado.mrt_paidsocial_pivot
GROUP BY channel, lob
HAVING SUM(CASE WHEN spend IS NULL THEN 1 ELSE 0 END) > 0
ORDER BY n_null_spend DESC;


SELECT
  qgp_date,
  channel_group,
  SUM(metric_value) AS spend,
  ROUND(AVG(wow_pct), 4) AS wow_pct
FROM prdrzranalytics.lab42.sdi_tbl_dashboardPulseTms_silver_biddableSpend_weekly
WHERE channel_group != 'All Channels'   -- avoid double-counting; this sums the 3 real channel_groups instead
  AND is_complete_period = true
GROUP BY qgp_date, channel_group
ORDER BY qgp_date DESC, channel_group;

SELECT
  qgp_date,
  lob,
  channel_group,
  metric_value AS spend,
  wow_pct
FROM prdrzranalytics.lab42.sdi_tbl_dashboardPulseTms_silver_biddableSpend_weekly
WHERE channel_group != 'All Channels'
  AND is_complete_period = true
ORDER BY qgp_date DESC, lob, channel_group;

SELECT
  week_sun_sat,
  channel_group,
  platform,
  SUM(spend) AS spend
FROM prdrzranalytics.lab42.sdi_tbl_dashboardPulseTms_bronze_biddableSpend_weekly
GROUP BY week_sun_sat, channel_group, platform
ORDER BY week_sun_sat DESC, channel_group, spend DESC;
