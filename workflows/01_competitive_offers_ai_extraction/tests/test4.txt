-- Broader check across all historical boundaries, not just this one campaign/week
SELECT
  COUNT(*) AS remaining_orphans,
  SUM(stub.Spend_Forecast) AS remaining_orphan_dollars
FROM prdrzranalytics.lab42.sdi_tbl_mfc_silver_spendGranular_weekly stub
JOIN prdrzranalytics.lab42.sdi_tbl_mfc_silver_spendGranular_weekly first
  ON stub.Week_Beginning_Monday = first.Week_Beginning_Monday
 AND stub.LOB_Supported = first.LOB_Supported
 AND stub.Channel <=> first.Channel AND stub.Tactic <=> first.Tactic
 AND stub.Message_Type <=> first.Message_Type AND stub.Agency <=> first.Agency
WHERE stub.week_type = 'BOUNDARY_STUB'
  AND stub.Spend_Actual IS NULL AND stub.Spend_Forecast IS NOT NULL
  AND first.week_type = 'BOUNDARY_FIRST'
  AND first.Spend_Actual IS NOT NULL;
output:
remaining_orphans	remaining_orphan_dollars
0	null

SELECT
  ng.QGP_Week, ng.LOB_Supported,
  ng.Spend_Actual AS NonGranular_Actual, g.Spend_Actual AS Granular_Actual,
  ng.Spend_Forecast AS NonGranular_Forecast, g.Spend_Forecast AS Granular_Forecast
FROM (
  SELECT QGP_Week, LOB_Supported, Spend_Actual, Spend_Forecast
  FROM prdrzranalytics.lab42.sdi_tbl_mfc_silver_spend_weekly
) ng
JOIN (
  SELECT QGP_Week, LOB_Supported, SUM(Spend_Actual) AS Spend_Actual, SUM(Spend_Forecast) AS Spend_Forecast
  FROM prdrzranalytics.lab42.sdi_tbl_mfc_silver_spendGranular_weekly
  GROUP BY QGP_Week, LOB_Supported
) g
  ON ng.QGP_Week = g.QGP_Week AND ng.LOB_Supported = g.LOB_Supported
WHERE ABS(COALESCE(ng.Spend_Actual, 0) - COALESCE(g.Spend_Actual, 0)) > 0.01
   OR ABS(COALESCE(ng.Spend_Forecast, 0) - COALESCE(g.Spend_Forecast, 0)) > 0.01
ORDER BY ng.QGP_Week DESC;
