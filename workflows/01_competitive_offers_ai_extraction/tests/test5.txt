SELECT MetricID, WeekEnding, DateContext, Amount
FROM prdrzrlakehouse.qgp_restricted.qgpweeklyview
WHERE UPPER(TRIM(MetricID)) = 'VRINBOUNDCALLSINCLHSIAUTOMATEDMANUAL'
  AND UPPER(TRIM(MetricType)) = 'ACTUALS/OUTLOOK'
  AND UPPER(TRIM(Page)) = 'HERO - CORE POSTPAID KP2'
  AND WeekEnding BETWEEN '2026-07-04' AND '2026-09-30'
  AND DateContext IN ('Normal', 'QTD', 'QTR', 'Monthly Normal')
ORDER BY WeekEnding, DateContext;
