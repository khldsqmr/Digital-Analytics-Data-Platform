//c.tms.cb.Metric Label
IF [Data Source] = 'ADOBE' THEN
  CASE [Metric Name]
    WHEN 'upvTotalAdobe'             THEN 'UPV'
    WHEN 'upvFlowTotal'              THEN 'UPV Flow'
    WHEN 'upvPostpaid'               THEN 'UPV Flow (Postpaid)'
    WHEN 'upvHsi'                    THEN 'UPV Flow (HSI)'
    WHEN 'upvByod'                   THEN 'UPV Flow (BYOD)'
    WHEN 'cartstartTotal'            THEN 'Add to Cart'
    WHEN 'cartstartPostpaid'         THEN 'CartStart (Postpaid)'
    WHEN 'cartstartHsi'              THEN 'CartStart (HSI)'
    WHEN 'cartstartByod'             THEN 'CartStart (BYOD)'
    WHEN 'ordersUnassistedTotal'     THEN 'Orders Unassisted'
    WHEN 'ordersUnassistedPostpaid'  THEN 'Orders Unassisted (Postpaid)'
    WHEN 'ordersUnassistedHsi'       THEN 'Orders Unassisted (HSI)'
    WHEN 'ordersUnassistedByod'      THEN 'Orders Unassisted (BYOD)'
    WHEN 'ordersAssistedTotal'       THEN 'Orders Assisted'
    WHEN 'ordersAssistedPostpaid'    THEN 'Orders Assisted (Postpaid)'
    WHEN 'ordersAssistedHsi'         THEN 'Orders Assisted (HSI)'
    WHEN 'ordersAssistedByod'        THEN 'Orders Assisted (BYOD)'
    ELSE NULL
  END

ELSEIF [Data Source] = 'MFC_SPEND_CHANNEL' THEN
  IF [Metric Name] = 'mfcSpendActual' THEN
    IF [Lob] = 'POSTPAID'      THEN 'MFC Spend Actual (Postpaid)'
    ELSEIF [Lob] = 'BROADBAND' THEN 'MFC Spend Actual (Broadband)'
    ELSE NULL
    END
  ELSEIF [Metric Name] = 'mfcSpendForecast' THEN
    IF [Lob] = 'POSTPAID'      THEN 'MFC Spend Forecast (Postpaid)'
    ELSEIF [Lob] = 'BROADBAND' THEN 'MFC Spend Forecast (Broadband)'
    ELSE NULL
    END
  ELSE NULL
  END

ELSEIF [Data Source] = 'PLATFORM_SPEND_CHANNEL' THEN
  IF [Metric Name] = 'platformSpend' THEN
    IF [Lob] = 'POSTPAID'      THEN 'Platform Spend (Postpaid)'
    ELSEIF [Lob] = 'BROADBAND' THEN 'Platform Spend (Broadband)'
    ELSE NULL
    END
  ELSE NULL
  END

END
