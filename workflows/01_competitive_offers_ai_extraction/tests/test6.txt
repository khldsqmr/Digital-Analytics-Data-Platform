-- 2A. PAID SOCIAL: find agency-related column
SELECT
    column_name
FROM prdrzranalytics.information_schema.columns
WHERE table_schema = 'lab42'
  AND table_name = 'media_analytics_integrated_snapshot'
  AND LOWER(column_name) LIKE '%agency%'
ORDER BY column_name;
output:
column_name
Agency
Creative_Agency
Creative_Agency_Ad_Number

-- 2B. PAID SOCIAL: see agency values for Paid Social
SELECT
    Agency,
    COUNT(*) AS rows,
    SUM(TRY_CAST(Spend AS DOUBLE)) AS spend
FROM prdrzranalytics.lab42.media_analytics_integrated_snapshot
WHERE Channel_Group_Name = 'Paid Social'
  AND Date >= date_sub(current_date(), 35)
GROUP BY Agency
ORDER BY spend DESC;
