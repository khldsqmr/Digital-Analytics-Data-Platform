CREATE OR REPLACE VIEW
prdrzranalytics.lab42.sdi_vw_pipelineMonitoring_gold_bqUcAlertSummary_daily
AS

WITH latest_run AS (

    ------------------------------------------------------------
    -- 1. Identify the latest available BQ → UC pipeline run
    ------------------------------------------------------------
    SELECT
        run_date
    FROM
        prdrzranalytics.lab42.sdi_vw_pipelineMonitoring_gold_bqUcRunDetails_daily
    WHERE
        run_date IS NOT NULL
    ORDER BY
        run_date DESC,
        job_run_ts DESC NULLS LAST,
        job_end_ts DESC NULLS LAST
    LIMIT 1

),

health AS (

    ------------------------------------------------------------
    -- 2. Get overall pipeline health for the latest run
    ------------------------------------------------------------
    SELECT
        h.run_date,
        MAX(h.total_tables) AS total_tables,
        MAX(h.successful_tables) AS successful_tables

    FROM
        prdrzranalytics.lab42.sdi_vw_pipelineMonitoring_gold_bqUcHealthSummary_daily h

    INNER JOIN latest_run l
        ON h.run_date = l.run_date

    GROUP BY
        h.run_date

),

issues AS (

    ------------------------------------------------------------
    -- 3. Get all table-level issues for the latest run
    ------------------------------------------------------------
    SELECT
        i.*
    FROM
        prdrzranalytics.lab42.sdi_vw_pipelineMonitoring_gold_bqUcTableIssues_daily i

    INNER JOIN latest_run l
        ON i.run_date = l.run_date

),

issue_rollup AS (

    ------------------------------------------------------------
    -- 4. Aggregate issue counts + affected tables
    ------------------------------------------------------------
    SELECT

        COUNT(*) AS issue_record_count,

        COUNT(DISTINCT bq_table)
            AS affected_table_count,

        --------------------------------------------------------
        -- Issue Counts
        --------------------------------------------------------

        COUNT(
            DISTINCT CASE
                WHEN issue_type = 'Failed Table'
                THEN bq_table
            END
        ) AS failed_table_count,

        COUNT(
            DISTINCT CASE
                WHEN issue_type = 'Skipped Table'
                THEN bq_table
            END
        ) AS skipped_table_count,

        COUNT(
            DISTINCT CASE
                WHEN issue_type = 'Unknown Status'
                THEN bq_table
            END
        ) AS unknown_status_count,

        COUNT(
            DISTINCT CASE
                WHEN issue_type = 'Row-Count Mismatch'
                THEN bq_table
            END
        ) AS row_count_mismatch_count,

        COUNT(
            DISTINCT CASE
                WHEN issue_type = 'Schema Change'
                THEN bq_table
            END
        ) AS schema_change_count,

        --------------------------------------------------------
        -- Failed Tables
        --------------------------------------------------------

        CONCAT_WS(
            ', ',
            SORT_ARRAY(
                COLLECT_SET(
                    CASE
                        WHEN issue_type = 'Failed Table'
                        THEN bq_table
                    END
                )
            )
        ) AS failed_tables,

        --------------------------------------------------------
        -- Skipped Tables
        --------------------------------------------------------

        CONCAT_WS(
            ', ',
            SORT_ARRAY(
                COLLECT_SET(
                    CASE
                        WHEN issue_type = 'Skipped Table'
                        THEN bq_table
                    END
                )
            )
        ) AS skipped_tables,

        --------------------------------------------------------
        -- Unknown Status Tables
        --------------------------------------------------------

        CONCAT_WS(
            ', ',
            SORT_ARRAY(
                COLLECT_SET(
                    CASE
                        WHEN issue_type = 'Unknown Status'
                        THEN bq_table
                    END
                )
            )
        ) AS unknown_status_tables,

        --------------------------------------------------------
        -- Row-Count Mismatch Tables
        --------------------------------------------------------

        CONCAT_WS(
            ', ',
            SORT_ARRAY(
                COLLECT_SET(
                    CASE
                        WHEN issue_type = 'Row-Count Mismatch'
                        THEN bq_table
                    END
                )
            )
        ) AS row_count_mismatch_tables,

        --------------------------------------------------------
        -- Schema Change Tables
        --------------------------------------------------------

        CONCAT_WS(
            ', ',
            SORT_ARRAY(
                COLLECT_SET(
                    CASE
                        WHEN issue_type = 'Schema Change'
                        THEN bq_table
                    END
                )
            )
        ) AS schema_change_tables

    FROM issues

)

------------------------------------------------------------
-- 5. Final monitoring output
------------------------------------------------------------
SELECT

    l.run_date,

    ------------------------------------------------------------
    -- Alert Trigger
    ------------------------------------------------------------

    CASE
        WHEN COALESCE(i.issue_record_count, 0) > 0
        THEN 1
        ELSE 0
    END AS alert_trigger,

    ------------------------------------------------------------
    -- Overall Counts
    ------------------------------------------------------------

    COALESCE(h.total_tables, 0)
        AS total_tables,

    COALESCE(h.successful_tables, 0)
        AS successful_tables,

    COALESCE(i.affected_table_count, 0)
        AS affected_table_count,

    ------------------------------------------------------------
    -- Issue Counts
    ------------------------------------------------------------

    COALESCE(i.failed_table_count, 0)
        AS failed_table_count,

    COALESCE(i.skipped_table_count, 0)
        AS skipped_table_count,

    COALESCE(i.unknown_status_count, 0)
        AS unknown_status_count,

    COALESCE(i.row_count_mismatch_count, 0)
        AS row_count_mismatch_count,

    COALESCE(i.schema_change_count, 0)
        AS schema_change_count,

    ------------------------------------------------------------
    -- Issue Table Lists
    ------------------------------------------------------------

    i.failed_tables,
    i.skipped_tables,
    i.unknown_status_tables,
    i.row_count_mismatch_tables,
    i.schema_change_tables

FROM latest_run l

LEFT JOIN health h
    ON h.run_date = l.run_date

CROSS JOIN issue_rollup i;
