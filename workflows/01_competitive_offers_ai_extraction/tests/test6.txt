// c.Show Channel Group by Spend Source

IF CONTAINS(UPPER([p.tms.Spend Source]), "MFC") THEN

    [Channel Group] IN (
        "Paid Search",
        "Programmatic",
        "Other",
        "Paid Social"
    )

ELSEIF CONTAINS(UPPER([p.tms.Spend Source]), "PLATFORM") THEN

    [Channel Group] IN (
        "Other",
        "Paid Search",
        "Programmatic",
        "Paid Social",
        "Affiliate",
        "iSpot National TV",
        "iSpot OTT"
    )

ELSEIF CONTAINS(UPPER([p.tms.Spend Source]), "BIDDABLE") THEN

    [Channel Group] IN (
        "Paid Search",
        "Programmatic",
        "Paid Social"
    )

ELSE
    FALSE
END
