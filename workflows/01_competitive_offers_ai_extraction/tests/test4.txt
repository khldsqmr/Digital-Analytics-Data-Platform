-- 1. Confirm Bronze mais spend has recent data at all
SELECT event_date, COUNT(*) AS rows, SUM(mais_platform_spend) AS total_spend
FROM `prj-dbi-prd-1.ds_dbi_digitalmedia_automation.vw_sdi_tsd_bronze_maisSpend_daily`
WHERE event_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 30 DAY)
GROUP BY event_date ORDER BY event_date;

-- 2. Check the real lob values against the hardcoded 'POSTPAID' filter
SELECT lob, COUNT(*) AS rows, SUM(mais_platform_spend) AS spend
FROM `prj-dbi-prd-1.ds_dbi_digitalmedia_automation.vw_sdi_tsd_bronze_maisSpend_daily`
GROUP BY lob
ORDER BY lob;

-- 3. Check the real Paid-Search-adjacent channel values against the 4 hardcoded literals
SELECT channel, COUNT(*) AS rows, SUM(mais_platform_spend) AS spend
FROM `prj-dbi-prd-1.ds_dbi_digitalmedia_automation.vw_sdi_tsd_bronze_maisSpend_daily`
WHERE lob = 'POSTPAID'
  AND channel LIKE '%PAID SEARCH%'
GROUP BY channel
ORDER BY channel;

-- 4. Confirm what Silver actually outputs for Paid Search once 1-3 are validated
SELECT event_date, lob, channel,
       mais_platform_spend, mais_platform_spend_branded, mais_platform_spend_nonbranded,
       mais_platform_spend_pla, mais_platform_spend_pmax
FROM `prj-dbi-prd-1.ds_dbi_digitalmedia_automation.vw_sdi_tsd_silver_maisSpend_daily`
WHERE channel = 'PAID SEARCH'
  AND event_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 30 DAY)
ORDER BY event_date;
