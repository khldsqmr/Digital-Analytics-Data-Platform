WITH actual_summary AS (
  SELECT QGP_Week, LOB_Supported, Channel, Tactic, Message_Type, Agency,
    COUNT(DISTINCT Quarter) AS n_quarters,
    MAX(Quarter) AS the_quarter,
    SUM(weekly_actual) AS total_actual
  FROM `prj-dbi-prd-1.ds_dbi_digitalmedia_automation.sdi_mfc_bronze_spendActualsGranular_weekly`
  GROUP BY 1,2,3,4,5,6
),
forecast_summary AS (
  SELECT QGP_Week, LOB_Supported, Channel, Tactic, Message_Type, Agency,
    COUNT(DISTINCT Quarter) AS n_quarters,
    MAX(Quarter) AS the_quarter,
    SUM(weekly_forecast) AS total_forecast
  FROM `prj-dbi-prd-1.ds_dbi_digitalmedia_automation.sdi_mfc_bronze_spendForecastGranular_weekly`
  GROUP BY 1,2,3,4,5,6
)
SELECT
  COUNT(*) AS genuine_mismatched_combos,
  SUM(f.total_forecast) AS genuine_orphaned_forecast_dollars,
  MIN(COALESCE(a.QGP_Week, f.QGP_Week)) AS earliest_week,
  MAX(COALESCE(a.QGP_Week, f.QGP_Week)) AS latest_week
FROM actual_summary a
FULL OUTER JOIN forecast_summary f
  ON a.QGP_Week = f.QGP_Week AND a.LOB_Supported = f.LOB_Supported
 AND a.Channel IS NOT DISTINCT FROM f.Channel AND a.Tactic IS NOT DISTINCT FROM f.Tactic
 AND a.Message_Type IS NOT DISTINCT FROM f.Message_Type AND a.Agency IS NOT DISTINCT FROM f.Agency
WHERE a.n_quarters = 1 AND f.n_quarters = 1 AND a.the_quarter != f.the_quarter;
