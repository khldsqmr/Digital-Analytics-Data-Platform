SELECT MetricID, WeekEnding, MetricType, Page, DateContext, DisplayMetricType, IsFuture, Amount
FROM prdrzrlakehouse.qgp_restricted.qgpweeklyview
WHERE UPPER(TRIM(MetricID)) IN (
  'CONSUMERPOSTPAIDBTSBOPISUNASSISTEDACTIVATIONSTM1MAPPEDDIGITAL',
  'CONSUMERPOSTPAIDBTSNONBOPISUNASSISTEDACTIVATIONSTM1MAPPEDDIGITAL'
)
AND UPPER(TRIM(DateContext)) = 'NORMAL'
AND UPPER(TRIM(Page)) = 'DIGITAL TRANSFORMATION NEW'
ORDER BY MetricID, WeekEnding DESC;
