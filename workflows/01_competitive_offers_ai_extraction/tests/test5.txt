%python

# ==================================================================================================
# NOTEBOOK:       sdi_nb_pipelineMonitoring_bqUcHealthCheck_daily
# NOTEBOOK TYPE:  Databricks Python Notebook
#
# PURPOSE:
#   Monitor daily BigQuery -> Unity Catalog pipeline health.
#
# CATALOG.SCHEMA:
#   prdrzranalytics.lab42
#
# SOURCE VIEWS:
#   sdi_vw_pipelineMonitoring_gold_bqUcRunDetails_daily
#   sdi_vw_pipelineMonitoring_gold_bqUcHealthSummary_daily
#   sdi_vw_pipelineMonitoring_gold_bqUcTableIssues_daily
#
# ISSUE CATEGORIES:
#   1. Failed Table
#   2. Unknown Status
#   3. Row-Count Mismatch
#   4. Schema Change
#   5. Skipped Table
#
# EXECUTION BEHAVIOR:
#   - Determines the latest available pipeline run date.
#   - Retrieves the health summary for that date.
#   - Retrieves all table issues for that date.
#   - Excludes healthy tables from detailed issue reporting.
#   - Allows one table to appear under multiple issue categories.
#   - Completes successfully when no issues are detected.
#   - Raises RuntimeError when one or more issues are detected.
#   - The RuntimeError causes the Databricks Workflow task to fail.
#   - Databricks Workflow failure notification sends the MS Teams alert.
#
# DATABRICKS JOB:
#   sdi_job_pipelineMonitoring_bqUcHealthCheck_daily
#
# DATABRICKS TASK:
#   run_pipelineMonitoring_bqUcHealthCheck
#
# TEAMS NOTIFICATION DESTINATION:
#   Lab42_SDITeamsAlert
#
# PARAMETERS:
#   None
# ==================================================================================================


# --------------------------------------------------------------------------------------------------
# Monitoring views
# --------------------------------------------------------------------------------------------------

RUN_DETAILS_VIEW = (
    "prdrzranalytics.lab42."
    "sdi_vw_pipelineMonitoring_gold_bqUcRunDetails_daily"
)

HEALTH_SUMMARY_VIEW = (
    "prdrzranalytics.lab42."
    "sdi_vw_pipelineMonitoring_gold_bqUcHealthSummary_daily"
)

TABLE_ISSUES_VIEW = (
    "prdrzranalytics.lab42."
    "sdi_vw_pipelineMonitoring_gold_bqUcTableIssues_daily"
)


# --------------------------------------------------------------------------------------------------
# Published monitoring dashboard
#
# IMPORTANT:
#   Store this as a normal URL.
#   Do NOT use Markdown syntax such as:
#       [https://...](https://...)
# --------------------------------------------------------------------------------------------------

PIPELINE_DASHBOARD_URL = (
    "https://adb-3157519905744188.8.azuredatabricks.net/"
    "dashboardsv3/01f1a173c1941a8781307551adc9950e/published"
    "?o=3157519905744188"
)


# --------------------------------------------------------------------------------------------------
# Standard issue display order
# --------------------------------------------------------------------------------------------------

ISSUE_DISPLAY_ORDER = [
    "Failed Table",
    "Unknown Status",
    "Row-Count Mismatch",
    "Schema Change",
    "Skipped Table"
]


# --------------------------------------------------------------------------------------------------
# Alert-message configuration
#
# Limits the number of table names included in the RuntimeError message.
# This is a local Python constant and not a Databricks job parameter.
# --------------------------------------------------------------------------------------------------

MAX_TABLES_IN_MESSAGE = 20


# --------------------------------------------------------------------------------------------------
# Validate configuration
# --------------------------------------------------------------------------------------------------

required_configuration = {
    "RUN_DETAILS_VIEW": RUN_DETAILS_VIEW,
    "HEALTH_SUMMARY_VIEW": HEALTH_SUMMARY_VIEW,
    "TABLE_ISSUES_VIEW": TABLE_ISSUES_VIEW,
    "PIPELINE_DASHBOARD_URL": PIPELINE_DASHBOARD_URL
}


for configuration_name, configuration_value in required_configuration.items():

    if (
        configuration_value is None
        or not str(configuration_value).strip()
    ):
        raise ValueError(
            "PIPELINE MONITORING CONFIGURATION ERROR: "
            f"{configuration_name} is blank."
        )


if not isinstance(MAX_TABLES_IN_MESSAGE, int):
    raise TypeError(
        "PIPELINE MONITORING CONFIGURATION ERROR: "
        "MAX_TABLES_IN_MESSAGE must be an integer."
    )


if MAX_TABLES_IN_MESSAGE <= 0:
    raise ValueError(
        "PIPELINE MONITORING CONFIGURATION ERROR: "
        "MAX_TABLES_IN_MESSAGE must be greater than zero."
    )


# --------------------------------------------------------------------------------------------------
# Print initialization information
# --------------------------------------------------------------------------------------------------

separator = "=" * 100


print(separator)
print("SDI BQ -> UC PIPELINE MONITORING")
print(separator)

print(f"Run details view      : {RUN_DETAILS_VIEW}")
print(f"Health summary view   : {HEALTH_SUMMARY_VIEW}")
print(f"Table issues view     : {TABLE_ISSUES_VIEW}")
print(f"Monitoring dashboard : {PIPELINE_DASHBOARD_URL}")

print("\nDetermining latest available pipeline run date...")


# --------------------------------------------------------------------------------------------------
# Determine the latest pipeline run date.
#
# Sorting by:
#   1. run_date
#   2. job_run_ts
#   3. job_end_ts
#
# ensures the timestamp displayed belongs to the latest available execution.
# --------------------------------------------------------------------------------------------------

latest_run_df = spark.sql(
    f"""
    SELECT
        run_date AS latest_run_date,
        job_run_ts AS latest_job_run_ts,
        job_end_ts AS latest_job_end_ts
    FROM {RUN_DETAILS_VIEW}
    WHERE run_date IS NOT NULL
    ORDER BY
        run_date DESC,
        job_run_ts DESC NULLS LAST,
        job_end_ts DESC NULLS LAST
    LIMIT 1
    """
)


latest_run_row = latest_run_df.first()


# --------------------------------------------------------------------------------------------------
# Validate that pipeline execution data exists.
# --------------------------------------------------------------------------------------------------

if latest_run_row is None:
    raise RuntimeError(
        "PIPELINE MONITORING ERROR: "
        f"No pipeline execution records were found in {RUN_DETAILS_VIEW}."
    )


latest_run_date = latest_run_row["latest_run_date"]
latest_job_run_ts = latest_run_row["latest_job_run_ts"]
latest_job_end_ts = latest_run_row["latest_job_end_ts"]


if latest_run_date is None:
    raise RuntimeError(
        "PIPELINE MONITORING ERROR: "
        f"The latest run date in {RUN_DETAILS_VIEW} is null."
    )


# Convert to YYYY-MM-DD text for use in SQL filters and messages.
latest_run_date_sql = str(latest_run_date)


# --------------------------------------------------------------------------------------------------
# Print selected monitoring date
# --------------------------------------------------------------------------------------------------

print(f"\nLatest pipeline date      : {latest_run_date}")
print(f"Latest execution start    : {latest_job_run_ts}")
print(f"Latest execution end      : {latest_job_end_ts}")

print(
    "\nMonitoring will evaluate the latest execution "
    f"of every table for {latest_run_date}."
)

print(separator)

%python

# ==================================================================================================
# CELL 3
#
# PURPOSE:
#   Retrieve all table issues for the latest pipeline run date.
#
# INCLUDED ISSUE CATEGORIES:
#   - Failed Table
#   - Unknown Status
#   - Row-Count Mismatch
#   - Schema Change
#   - Skipped Table
#
# IMPORTANT:
#   - Healthy tables are excluded by the underlying table-issues view.
#   - One table can appear under multiple issue categories.
#   - Any returned issue record causes the monitoring result to FAIL.
# ==================================================================================================


print("=" * 100)
print("TABLE ISSUE CHECK")
print("=" * 100)

print(f"Checking table issues for: {latest_run_date}")


# --------------------------------------------------------------------------------------------------
# Retrieve issue records.
# --------------------------------------------------------------------------------------------------

issues_df = spark.sql(
    f"""
    SELECT
        issue_type,
        run_date,
        job_run_ts,
        job_end_ts,
        batch_number,
        bq_table,
        uc_table,
        load_type,
        run_type,
        status,
        status_raw,
        is_failed,
        bq_rows,
        loaded_rows,
        row_count_difference,
        has_rowCountMismatch,
        elapsed_sec,
        error_message,
        cols_added,
        cols_removed,
        cols_changed,
        has_schemaChange
    FROM {TABLE_ISSUES_VIEW}
    WHERE run_date = DATE('{latest_run_date_sql}')
    ORDER BY
        CASE issue_type
            WHEN 'Failed Table' THEN 1
            WHEN 'Unknown Status' THEN 2
            WHEN 'Row-Count Mismatch' THEN 3
            WHEN 'Schema Change' THEN 4
            WHEN 'Skipped Table' THEN 5
            ELSE 6
        END,
        batch_number,
        bq_table,
        uc_table
    """
)


# --------------------------------------------------------------------------------------------------
# Collect issue records.
#
# Because TABLE_ISSUES_VIEW contains only exception records, any row returned
# here means that pipeline attention is required.
# --------------------------------------------------------------------------------------------------

issue_rows = issues_df.collect()

issue_record_count = len(issue_rows)

has_issues = issue_record_count > 0


# --------------------------------------------------------------------------------------------------
# Create normalized table/issue combinations.
#
# Prefer BQ table name.
# Fall back to UC table name if BQ table name is unavailable.
# --------------------------------------------------------------------------------------------------

normalized_issue_records = []


for row in issue_rows:

    issue_type = (
        row["issue_type"]
        if row["issue_type"] is not None
        else "Uncategorized Issue"
    )

    table_name = (
        row["bq_table"]
        if row["bq_table"] is not None
        else row["uc_table"]
    )

    if table_name is None:
        table_name = "UNKNOWN_TABLE"

    normalized_issue_records.append(
        (
            issue_type,
            table_name
        )
    )


# --------------------------------------------------------------------------------------------------
# Distinct affected tables.
# --------------------------------------------------------------------------------------------------

affected_table_names = list(
    dict.fromkeys(
        table_name
        for _, table_name in normalized_issue_records
    )
)

affected_table_count = len(affected_table_names)


# --------------------------------------------------------------------------------------------------
# Distinct issue-type / table combinations.
#
# Example:
#   table_a -> Schema Change
#   table_a -> Row-Count Mismatch
#
# counts as:
#   1 affected table
#   2 table-issue combinations
# --------------------------------------------------------------------------------------------------

issue_combinations = list(
    dict.fromkeys(
        normalized_issue_records
    )
)

issue_combination_count = len(issue_combinations)


# --------------------------------------------------------------------------------------------------
# Count distinct tables within each issue category.
# --------------------------------------------------------------------------------------------------

issue_counts = {
    issue_type: 0
    for issue_type in ISSUE_DISPLAY_ORDER
}


for issue_type, table_name in issue_combinations:

    if issue_type not in issue_counts:
        issue_counts[issue_type] = 0

    issue_counts[issue_type] += 1


# --------------------------------------------------------------------------------------------------
# Preserve the standard order first, followed by any unexpected issue types.
# --------------------------------------------------------------------------------------------------

unexpected_issue_types = sorted(
    issue_type
    for issue_type in issue_counts
    if issue_type not in ISSUE_DISPLAY_ORDER
    and issue_counts.get(issue_type, 0) > 0
)


issue_types_for_output = (
    ISSUE_DISPLAY_ORDER
    + unexpected_issue_types
)


# --------------------------------------------------------------------------------------------------
# Print issue summary.
# --------------------------------------------------------------------------------------------------

print(
    f"\nRaw issue records                 : "
    f"{issue_record_count}"
)

print(
    f"Distinct affected tables          : "
    f"{affected_table_count}"
)

print(
    f"Distinct table-issue combinations : "
    f"{issue_combination_count}"
)


# --------------------------------------------------------------------------------------------------
# No issues.
# --------------------------------------------------------------------------------------------------

if not has_issues:

    print("\nSUCCESS")

    print(
        "No BQ -> UC table issues were detected "
        "for the latest pipeline date."
    )


# --------------------------------------------------------------------------------------------------
# Issues detected.
# --------------------------------------------------------------------------------------------------

else:

    print("\nTABLE ISSUES DETECTED")
    print("-" * 100)


    for issue_type in issue_types_for_output:

        issue_count = issue_counts.get(
            issue_type,
            0
        )

        if issue_count > 0:

            print(
                f"{issue_type:<22}: "
                f"{issue_count}"
            )


    print("-" * 100)

    print("\nAffected tables:")


    for table_number, table_name in enumerate(
        affected_table_names,
        start=1
    ):

        print(
            f"{table_number:>2}. {table_name}"
        )


    print("-" * 100)


# --------------------------------------------------------------------------------------------------
# Display all detailed issue records.
# --------------------------------------------------------------------------------------------------

display(issues_df)







%python

# ==================================================================================================
# CELL 4
#
# PURPOSE:
#   Produce the final BQ -> UC pipeline monitoring result.
#
# SUCCESS:
#   - No table issues exist.
#   - Notebook completes normally.
#   - Databricks Workflow task is marked SUCCESS.
#
# FAILURE:
#   - One or more issue records exist.
#   - A concise monitoring error message is created.
#   - RuntimeError is intentionally raised.
#   - Databricks Workflow task is marked FAILED.
#   - Databricks sends the configured Microsoft Teams failure notification.
#
# REQUIRED DATABRICKS NOTIFICATION DESTINATION:
#   Lab42_SDITeamsAlert
#
# PARAMETERS:
#   None
# ==================================================================================================


separator = "=" * 100
sub_separator = "-" * 100


# --------------------------------------------------------------------------------------------------
# Validate variables produced by previous cells.
# --------------------------------------------------------------------------------------------------

required_variables = {
    "latest_run_date": latest_run_date,
    "latest_run_date_sql": latest_run_date_sql,
    "has_issues": has_issues,
    "issue_record_count": issue_record_count,
    "affected_table_names": affected_table_names,
    "affected_table_count": affected_table_count,
    "issue_combination_count": issue_combination_count,
    "issue_counts": issue_counts,
    "issue_types_for_output": issue_types_for_output,
    "TABLE_ISSUES_VIEW": TABLE_ISSUES_VIEW,
    "PIPELINE_DASHBOARD_URL": PIPELINE_DASHBOARD_URL,
    "MAX_TABLES_IN_MESSAGE": MAX_TABLES_IN_MESSAGE
}


for variable_name, variable_value in required_variables.items():

    if variable_value is None:

        raise RuntimeError(
            "PIPELINE MONITORING CONFIGURATION ERROR: "
            f"{variable_name} is not defined correctly."
        )


# --------------------------------------------------------------------------------------------------
# Print result header.
# --------------------------------------------------------------------------------------------------

print(separator)
print("BQ -> UC PIPELINE MONITORING RESULT")
print(separator)


# ==================================================================================================
# PASS
# ==================================================================================================

if not has_issues:

    print("Status                   : PASS")
    print(f"Run date                 : {latest_run_date}")
    print("Issue records            : 0")
    print("Affected table count     : 0")
    print("Table-issue combinations : 0")

    print(
        f"Monitoring view          : "
        f"{TABLE_ISSUES_VIEW}"
    )

    print(
        f"Monitoring dashboard     : "
        f"{PIPELINE_DASHBOARD_URL}"
    )


    print(sub_separator)
    print("RESULT")
    print(sub_separator)

    print(
        "No BQ -> UC table issues were detected "
        "for the monitored date."
    )

    print(separator)

    print(
        "SDI BQ -> UC pipeline monitoring "
        "completed successfully."
    )


# ==================================================================================================
# FAIL
# ==================================================================================================

else:

    table_label = (
        "table"
        if affected_table_count == 1
        else "tables"
    )

    issue_label = (
        "issue"
        if issue_combination_count == 1
        else "issues"
    )


    # ----------------------------------------------------------------------------------------------
    # Print overall failure status.
    # ----------------------------------------------------------------------------------------------

    print("Status                   : FAIL")
    print(f"Run date                 : {latest_run_date}")
    print(f"Issue records            : {issue_record_count}")
    print(f"Affected table count     : {affected_table_count}")

    print(
        f"Table-issue combinations : "
        f"{issue_combination_count}"
    )

    print(
        f"Monitoring view          : "
        f"{TABLE_ISSUES_VIEW}"
    )

    print(
        f"Monitoring dashboard     : "
        f"{PIPELINE_DASHBOARD_URL}"
    )


    # ----------------------------------------------------------------------------------------------
    # Print issue summary.
    # ----------------------------------------------------------------------------------------------

    print(sub_separator)
    print("ISSUE SUMMARY")
    print(sub_separator)


    for issue_type in issue_types_for_output:

        issue_count = issue_counts.get(
            issue_type,
            0
        )

        if issue_count > 0:

            print(
                f"{issue_type:<22}: "
                f"{issue_count}"
            )


    # ----------------------------------------------------------------------------------------------
    # Print affected table names.
    # ----------------------------------------------------------------------------------------------

    print(sub_separator)
    print("AFFECTED TABLES")
    print(sub_separator)


    for table_number, table_name in enumerate(
        affected_table_names,
        start=1
    ):

        print(
            f"{table_number:>2}. {table_name}"
        )


    # ----------------------------------------------------------------------------------------------
    # Print overall monitoring result.
    # ----------------------------------------------------------------------------------------------

    print(sub_separator)
    print("RESULT")
    print(sub_separator)


    print(
        f"{issue_combination_count} {issue_label} "
        f"detected across "
        f"{affected_table_count} {table_label}."
    )


    print(
        f"Review {TABLE_ISSUES_VIEW} "
        "for detailed issue information."
    )


    # ----------------------------------------------------------------------------------------------
    # Limit table names included in the Workflow exception.
    # ----------------------------------------------------------------------------------------------

    alert_table_names = affected_table_names[
        :MAX_TABLES_IN_MESSAGE
    ]


    table_list_text = ", ".join(
        str(table_name)
        for table_name in alert_table_names
    )


    if affected_table_count > MAX_TABLES_IN_MESSAGE:

        remaining_count = (
            affected_table_count
            - MAX_TABLES_IN_MESSAGE
        )

        remaining_label = (
            "table"
            if remaining_count == 1
            else "tables"
        )

        table_list_text += (
            f", and {remaining_count} additional "
            f"{remaining_label}"
        )


    if not table_list_text:
        table_list_text = "Table names unavailable"


    # ----------------------------------------------------------------------------------------------
    # Build issue summary text.
    # ----------------------------------------------------------------------------------------------

    issue_summary_parts = []


    for issue_type in issue_types_for_output:

        issue_count = issue_counts.get(
            issue_type,
            0
        )

        if issue_count > 0:

            issue_summary_parts.append(
                f"{issue_type}: {issue_count}"
            )


    issue_summary_text = "; ".join(
        issue_summary_parts
    )


    if not issue_summary_text:
        issue_summary_text = "Issue details unavailable"


    # ----------------------------------------------------------------------------------------------
    # Build final Workflow failure message.
    #
    # IMPORTANT:
    #   Keep this relatively concise because it becomes part of the Databricks
    #   task failure information.
    # ----------------------------------------------------------------------------------------------

    error_message = (
        "BQ -> UC PIPELINE ALERT | "
        "Status: FAIL | "
        f"Run Date: {latest_run_date} | "
        f"Affected Tables: {affected_table_count} | "
        f"Table-Issue Combinations: {issue_combination_count} | "
        f"Issue Summary: {issue_summary_text} | "
        f"Tables: {table_list_text} | "
        f"Dashboard: {PIPELINE_DASHBOARD_URL} | "
        f"Details View: {TABLE_ISSUES_VIEW}"
    )


    # ----------------------------------------------------------------------------------------------
    # Validate dashboard URL is included in the failure information.
    # ----------------------------------------------------------------------------------------------

    if PIPELINE_DASHBOARD_URL not in error_message:

        raise RuntimeError(
            "PIPELINE MONITORING CONFIGURATION ERROR: "
            "Monitoring dashboard URL is missing from "
            "the Workflow failure message."
        )


    # ----------------------------------------------------------------------------------------------
    # Print the exact failure information that will appear in the task output.
    # ----------------------------------------------------------------------------------------------

    print(sub_separator)
    print("WORKFLOW FAILURE MESSAGE")
    print(sub_separator)

    print(error_message)

    print(separator)


    # ==============================================================================================
    # INTENTIONAL WORKFLOW FAILURE
    #
    # DO NOT REMOVE THIS.
    #
    # This is what marks the Databricks task FAILED.
    # The configured Microsoft Teams failure notification is then triggered.
    # ==============================================================================================

    raise RuntimeError(error_message)
