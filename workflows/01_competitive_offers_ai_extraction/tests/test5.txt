SELECT Header, Section, MetricOrder, MetricID, MetricName, DisplayMetricType
FROM prdrzrlakehouse.qgp_restricted.qgpweeklyview
WHERE Page = 'Digital'
  AND DateContext = 'Normal'
  AND MetricType IN ('Actuals', 'Actuals/Outlook')
ORDER BY Header, MetricOrder;


SELECT Header, Section, MetricOrder, MetricID, MetricName, DisplayMetricType
FROM prdrzrlakehouse.qgp_restricted.qgpweeklyview
WHERE Page IN ('IT Summary', 'IT Summary Outcomes')
  AND DateContext = 'Normal'
  AND MetricType IN ('Actuals', 'Actuals/Outlook')
ORDER BY Header, MetricOrder;


SELECT DISTINCT Page
FROM prdrzrlakehouse.qgp_restricted.qgpweeklyview
WHERE LOWER(Page) LIKE '%digital%'
ORDER BY Page;
