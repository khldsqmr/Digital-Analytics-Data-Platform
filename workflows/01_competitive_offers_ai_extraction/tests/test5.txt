WITH Dedup AS (
  SELECT *
  FROM prdrzrlakehouse.qgp_restricted.qgpweeklyview
  QUALIFY ROW_NUMBER() OVER (
    PARTITION BY WeekEnding, MetricID, DateContext, MetricType
    ORDER BY InsertDateTime DESC, PublishKey DESC
  ) = 1
)
SELECT
  WeekEnding AS qgp_date,

  SUM(CASE WHEN UPPER(TRIM(MetricType)) = 'ACTUALS'
            AND UPPER(TRIM(Page)) = 'DIGITAL TRANSFORMATION NEW'
            AND UPPER(TRIM(DateContext)) = 'NORMAL'
            AND UPPER(TRIM(MetricID)) IN (
              'CONSUMERPOSTPAIDNEWPHONEBANBOPISUNASSISTEDACTIVATIONSTM1MAPPEDDIGITAL',
              'CONSUMERPOSTPAIDNEWPHONEBANNONBOPISUNASSISTEDACTIVATIONSTM1MAPPEDDIGITAL'
            )
       THEN Amount END)                                                   AS activationsBopis_actual,
  SUM(CASE WHEN UPPER(TRIM(MetricType)) = 'QGP'
            AND UPPER(TRIM(Page)) = 'DIGITAL TRANSFORMATION NEW'
            AND UPPER(TRIM(DateContext)) = 'NORMAL'
            AND UPPER(TRIM(MetricID)) = 'CONSUMERPOSTPAIDNEWPHONEBANACTIVATIONSTM1MAPPEDDIGITAL'
       THEN Amount END)                                                   AS activationsBopis_target,

  SUM(CASE WHEN CONTAINS(LOWER(TRIM(MetricName)), 'activations (new + aal')
            AND CONTAINS(LOWER(TRIM(MetricName)), 'no assistance')
            AND UPPER(TRIM(MetricType)) IN ('ACTUALS', 'ACTUALS/OUTLOOK')
            AND UPPER(TRIM(MetricID)) = 'TOTALDIGITALPHONEACTIVATIONSTMOANDSPRINTGLANCETM1MAPPED'
            AND UPPER(TRIM(Page)) = 'DIGITAL TRANSFORMATION NEW'
            AND LOWER(REPLACE(TRIM(IsFuture), ' ', '')) = 'ispast'
            AND UPPER(TRIM(DateContext)) = 'NORMAL'
       THEN Amount END)                                                   AS activationsNewAalNoAssistance_actual,
  SUM(CASE WHEN CONTAINS(LOWER(TRIM(MetricName)), 'activations (new + aal')
            AND CONTAINS(LOWER(TRIM(MetricName)), 'no assistance')
            AND UPPER(TRIM(MetricType)) = 'QGP'
            AND UPPER(TRIM(MetricID)) = 'TOTALDIGITALPHONEACTIVATIONSTMOANDSPRINTGLANCETM1MAPPED'
            AND UPPER(TRIM(Page)) = 'DIGITAL TRANSFORMATION NEW'
            AND UPPER(TRIM(DateContext)) = 'NORMAL'
       THEN Amount END)                                                   AS activationsNewAalNoAssistance_target,

  SUM(CASE WHEN LOWER(TRIM(MetricName)) = 'store traffic (excl store-in-store)'
            AND UPPER(TRIM(MetricType)) = 'ACTUALS/OUTLOOK'
            AND UPPER(TRIM(DateContext)) = 'NORMAL'
            AND LOWER(REPLACE(TRIM(IsFuture), ' ', '')) = 'ispast'
       THEN Amount END)                                                   AS storeTraffic_actual,
  SUM(CASE WHEN LOWER(TRIM(MetricName)) = 'store traffic (excl store-in-store)'
            AND UPPER(TRIM(MetricType)) = 'QGP'
            AND UPPER(TRIM(DateContext)) = 'NORMAL'
       THEN Amount END)                                                   AS storeTraffic_target,

  SUM(CASE WHEN UPPER(TRIM(MetricType)) = 'ACTUALS/OUTLOOK'
            AND UPPER(TRIM(MetricID)) = 'VRINBOUNDCALLSINCLHSIAUTOMATEDMANUAL'
            AND UPPER(TRIM(DateContext)) = 'NORMAL'
            AND LOWER(REPLACE(TRIM(IsFuture), ' ', '')) = 'ispast'
       THEN Amount END)                                                   AS vrCalls_actual,
  SUM(CASE WHEN UPPER(TRIM(MetricType)) = 'QGP'
            AND UPPER(TRIM(MetricID)) = 'VRINBOUNDCALLSINCLHSIAUTOMATEDMANUAL'
            AND UPPER(TRIM(DisplayMetricType)) = 'TARGET'
            AND UPPER(TRIM(DateContext)) = 'NORMAL'
            AND UPPER(TRIM(Page)) = 'VIRTUAL RETAIL'
       THEN Amount END)                                                   AS vrCalls_target,

  SUM(CASE WHEN UPPER(TRIM(MetricType)) = 'ACTUALS/OUTLOOK'
            AND UPPER(TRIM(MetricID)) = 'VRCHATSINCLHSIAUTOMATEDMANUAL'
            AND UPPER(TRIM(DateContext)) = 'NORMAL'
            AND LOWER(REPLACE(TRIM(IsFuture), ' ', '')) = 'ispast'
       THEN Amount END)                                                   AS vrChats_actual,
  SUM(CASE WHEN UPPER(TRIM(MetricType)) = 'QGP'
            AND UPPER(TRIM(MetricID)) = 'VRCHATSINCLHSIAUTOMATEDMANUAL'
            AND UPPER(TRIM(DisplayMetricType)) = 'TARGET'
            AND UPPER(TRIM(DateContext)) = 'NORMAL'
            AND UPPER(TRIM(Page)) = 'VIRTUAL RETAIL'
       THEN Amount END)                                                   AS vrChats_target,

  SUM(CASE WHEN CONTAINS(LOWER(TRIM(MetricName)), 'postpaid activations')
            AND UPPER(TRIM(MetricType)) = 'ACTUALS/OUTLOOK'
            AND UPPER(TRIM(MetricID)) = 'VRPOSTPAIDACTIVATIONSINCLVIRTUALBUSINESSTM1MAPPED'
            AND UPPER(TRIM(Page)) = 'VIRTUAL RETAIL OUTCOMES 1'
            AND LOWER(REPLACE(TRIM(IsFuture), ' ', '')) = 'ispast'
            AND UPPER(TRIM(DateContext)) = 'NORMAL'
       THEN Amount END)                                                   AS vrPostpaidActivations_actual,
  SUM(CASE WHEN CONTAINS(LOWER(TRIM(MetricName)), 'postpaid activations')
            AND UPPER(TRIM(MetricType)) = 'QGP'
            AND UPPER(TRIM(MetricID)) = 'VRPOSTPAIDACTIVATIONSINCLVIRTUALBUSINESSTM1MAPPED'
            AND UPPER(TRIM(Page)) = 'VIRTUAL RETAIL OUTCOMES 1'
            AND UPPER(TRIM(DateContext)) = 'NORMAL'
       THEN Amount END)                                                   AS vrPostpaidActivations_target,

  SUM(CASE WHEN UPPER(TRIM(Page)) = 'DIGITAL TRANSFORMATION NEW'
            AND UPPER(TRIM(MetricType)) IN ('ACTUALS', 'ACTUALS/OUTLOOK')
            AND UPPER(TRIM(DateContext)) = 'NORMAL'
            AND UPPER(TRIM(MetricID)) = 'CONPOSTPAIDDIGITALPCTOFNEWBANPHONEUNASSISTEDANDASSISTEDWONRINDIRECT'
            AND LOWER(REPLACE(TRIM(IsFuture), ' ', '')) = 'ispast'
       THEN Amount END)                                                   AS digitalPctPhoneNewActsNoAssistPlusAssist_actual,
  SUM(CASE WHEN UPPER(TRIM(Page)) = 'DIGITAL TRANSFORMATION NEW'
            AND UPPER(TRIM(MetricType)) = 'QGP'
            AND UPPER(TRIM(DateContext)) = 'NORMAL'
            AND UPPER(TRIM(MetricID)) = 'CONPOSTPAIDDIGITALPCTOFNEWBANPHONEUNASSISTEDANDASSISTEDWONRINDIRECT'
       THEN Amount END)                                                   AS digitalPctPhoneNewActsNoAssistPlusAssist_target,

  SUM(CASE WHEN UPPER(TRIM(MetricType)) = 'ACTUALS/OUTLOOK'
            AND UPPER(TRIM(MetricID)) = 'DIGITALPCTOFCONSUMERPOSTPAIDACTIVATIONSTOTALINCLASSISTEDNEW'
            AND UPPER(TRIM(DateContext)) = 'NORMAL'
            AND LOWER(REPLACE(TRIM(IsFuture), ' ', '')) = 'ispast'
            AND UPPER(TRIM(Page)) = 'DIGITAL'
       THEN Amount END)                                                   AS digitalPctConsumerPostpaidActivationsTotalInclAssisted_actual,
  SUM(CASE WHEN UPPER(TRIM(MetricType)) = 'QGP'
            AND UPPER(TRIM(MetricID)) = 'DIGITALPCTOFCONSUMERPOSTPAIDACTIVATIONSTOTALINCLASSISTEDNEW'
            AND UPPER(TRIM(Page)) = 'DIGITAL'
            AND UPPER(TRIM(DisplayMetricType)) = 'TARGET'
            AND UPPER(TRIM(DateContext)) = 'NORMAL'
       THEN Amount END)                                                   AS digitalPctConsumerPostpaidActivationsTotalInclAssisted_target,

  SUM(CASE WHEN UPPER(TRIM(Page)) = 'DIGITAL TRANSFORMATION NEW'
            AND UPPER(TRIM(MetricType)) IN ('ACTUALS', 'ACTUALS/OUTLOOK')
            AND UPPER(TRIM(DateContext)) = 'NORMAL'
            AND UPPER(TRIM(MetricID)) = 'DIGITALPCTOFCONSUMERPOSTPAIDACTIVATIONSPHONENEWSAMEPAGEWONRINDIRECT'
       THEN Amount END)                                                   AS digitalPctNoAssistanceActivations_actual,
  CAST(NULL AS DOUBLE)                                                    AS digitalPctNoAssistanceActivations_target,  -- confirmed no Target row exists

  SUM(CASE WHEN UPPER(TRIM(Page)) = 'DIGITAL TRANSFORMATION NEW'
            AND UPPER(TRIM(MetricType)) IN ('ACTUALS', 'ACTUALS/OUTLOOK')
            AND UPPER(TRIM(DateContext)) = 'NORMAL'
            AND UPPER(TRIM(MetricID)) = 'NEWPHONEBANASSISTEDACTIVATIONSDIGITALPCTOFCONSUMERPOSTPAIDTM1MAPPEDWONRINDIRECT'
       THEN Amount END)                                                   AS digitalPctAssistanceActivations_actual,
  CAST(NULL AS DOUBLE)                                                    AS digitalPctAssistanceActivations_target  -- confirmed no Target row exists

FROM Dedup
GROUP BY WeekEnding
ORDER BY WeekEnding DESC;
