SELECT
  lob,
  channel_group,
  COUNT(*) AS row_count,
  SUM(spend) AS spend
FROM prdrzranalytics.lab42.sdi_tbl_dashboardPulseTms_bronze_biddableSpend_weekly
WHERE week_sun_sat >= date_sub(current_date(), 35)
  AND lob IN ('POSTPAID', 'HSI', 'BROADBAND')
GROUP BY
  lob,
  channel_group
ORDER BY
  lob,
  channel_group;


SELECT
  lob,
  channel_group,
  COUNT(*) AS row_count,
  SUM(metric_value) AS spend
FROM prdrzranalytics.lab42.sdi_tbl_dashboardPulseTms_silver_biddableSpend_weekly
WHERE qgp_date >= date_sub(current_date(), 35)
  AND lob IN ('POSTPAID', 'BROADBAND')
  AND metric_name = 'biddableSpend'
GROUP BY
  lob,
  channel_group
ORDER BY
  lob,
  channel_group;

SELECT
  lob,
  channel_group,
  COUNT(*) AS row_count,
  SUM(metric_value) AS spend
FROM prdrzranalytics.lab42.sdi_vw_dashboardPulseTms_gold_unified_long
WHERE data_source = 'BIDDABLE_SPEND_CHANNEL'
  AND metric_name = 'biddableSpend'
  AND qgp_date >= date_sub(current_date(), 35)
GROUP BY
  lob,
  channel_group
ORDER BY
  lob,
  channel_group;

SELECT
  qgp_date,
  lob,
  channel_group,
  metric_value
FROM prdrzranalytics.lab42.sdi_vw_dashboardPulseTms_gold_unified_long
WHERE data_source = 'BIDDABLE_SPEND_CHANNEL'
  AND metric_name = 'biddableSpend'
  AND lob IN ('POSTPAID', 'BROADBAND')
ORDER BY
  qgp_date DESC,
  lob,
  CASE channel_group
    WHEN 'All Channels' THEN 1
    WHEN 'Programmatic' THEN 2
    WHEN 'Paid Search'  THEN 3
    WHEN 'Paid Social'  THEN 4
    ELSE 5
  END;
