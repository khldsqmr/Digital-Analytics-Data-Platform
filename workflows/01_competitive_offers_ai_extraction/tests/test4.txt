SELECT
  Quarter,
  LENGTH(Quarter) AS char_length,
  Quarter RLIKE "^Q[1-4]'[0-9]{2}$" AS matches_bronze_regex,
  COUNT(*) AS rows
FROM prdrzranalytics.lab42.raw_media_flowchart
WHERE UPPER(TRIM(LOB_Supported)) = 'CONSUMER POSTPAID'
  AND UPPER(TRIM(QGP)) = 'FORECAST'
  AND TRY_CAST(QGP_Week AS DATE) = '2026-01-03'
  AND TRY_CAST(CAST(FileLoad_Date AS STRING) AS DATE) = '2025-12-26'
GROUP BY Quarter, char_length, matches_bronze_regex;
