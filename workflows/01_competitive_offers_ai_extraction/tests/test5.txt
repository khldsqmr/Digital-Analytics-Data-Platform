SELECT MetricID, WeekEnding, MetricType, Page, COUNT(DISTINCT PublishKey) AS n_publish_keys, COUNT(DISTINCT Amount) AS n_distinct_amounts, MIN(Amount) AS min_amt, MAX(Amount) AS max_amt
FROM prdrzrlakehouse.qgp_restricted.qgpweeklyview
WHERE UPPER(TRIM(MetricID)) IN (
  'CONSUMERPOSTPAIDNEWPHONEBANBOPISUNASSISTEDACTIVATIONSTM1MAPPEDDIGITAL',
  'CONSUMERPOSTPAIDNEWPHONEBANNONBOPISUNASSISTEDACTIVATIONSTM1MAPPEDDIGITAL',

  'VRINBOUNDCALLSINCLHSIAUTOMATEDMANUAL',
