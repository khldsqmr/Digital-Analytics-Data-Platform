SELECT 
  view_source,
  Channel_Group_Abbr,
  Channel_Group_Name,
  Channel_Abbr,
  Channel_Name,
  COUNT(*) AS n,
  SUM(Spend) AS total_spend,
  MIN(Date) AS min_date,
  MAX(Date) AS max_date
FROM prdrzranalytics.lab42.media_analytics_integrated_snapshot
GROUP BY view_source, Channel_Group_Abbr, Channel_Group_Name, Channel_Abbr, Channel_Name
ORDER BY view_source, total_spend DESC;


SELECT 
  LOB,
  COUNT(*) AS n,
  SUM(Spend) AS total_spend,
  MIN(Date) AS min_date,
  MAX(Date) AS max_date
FROM prdrzranalytics.lab42.media_analytics_integrated_snapshot
GROUP BY LOB
ORDER BY total_spend DESC;


SELECT 
  COUNT(*) AS total_rows,
  COUNT(DISTINCT CONCAT_WS('|', CAST(Date AS STRING), Account_ID, Campaign_ID, Placement_ID, Creative_ID)) AS distinct_natural_key
FROM prdrzranalytics.lab42.media_analytics_integrated_snapshot;
