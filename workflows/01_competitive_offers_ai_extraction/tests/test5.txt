WITH params AS (
    SELECT
        date_sub(next_day(current_date(), 'SUN'), 8) AS prior_week_end,
        date_sub(next_day(current_date(), 'SUN'), 14) AS prior_week_start
)

SELECT
    p.prior_week_start,
    p.prior_week_end,

    MAX(s.Date) AS latest_date_available,

    DATEDIFF(
        current_date(),
        MAX(s.Date)
    ) AS data_lag_days,

    COUNT(DISTINCT CASE
        WHEN s.Date BETWEEN p.prior_week_start AND p.prior_week_end
        THEN s.Date
    END) AS prior_week_days_present,

    SUM(CASE
        WHEN s.Date BETWEEN p.prior_week_start AND p.prior_week_end
        THEN s.Spend
        ELSE 0
    END) AS prior_week_spend,

    CASE
        WHEN COUNT(DISTINCT CASE
                 WHEN s.Date BETWEEN p.prior_week_start AND p.prior_week_end
                 THEN s.Date
             END) = 7
        THEN 'PASS - all 7 days available'
        ELSE 'CHECK - prior week incomplete'
    END AS availability_status

FROM prdrzranalytics.lab42.media_analytics_integrated_snapshot s
CROSS JOIN params p

WHERE s.Channel_Group_Name = 'Paid Social'

GROUP BY
    p.prior_week_start,
    p.prior_week_end;

WITH params AS (
    SELECT
        date_sub(next_day(current_date(), 'SUN'), 8) AS prior_week_end,
        date_sub(next_day(current_date(), 'SUN'), 14) AS prior_week_start
),

social AS (
    SELECT
        CASE
            WHEN Channel_Name IN (
                'Paid Social - Facebook',
                'Paid Social - Instagram'
            ) THEN 'Meta'
            ELSE REPLACE(Channel_Name, 'Paid Social - ', '')
        END AS platform,

        Date,
        Spend

    FROM prdrzranalytics.lab42.media_analytics_integrated_snapshot
    WHERE Channel_Group_Name = 'Paid Social'
)

SELECT
    platform,

    MAX(Date) AS latest_date_available,

    COUNT(DISTINCT CASE
        WHEN Date BETWEEN p.prior_week_start AND p.prior_week_end
        THEN Date
    END) AS days_present_prior_week,

    SUM(CASE
        WHEN Date BETWEEN p.prior_week_start AND p.prior_week_end
        THEN Spend
        ELSE 0
    END) AS prior_week_spend,

    CASE
        WHEN COUNT(DISTINCT CASE
                 WHEN Date BETWEEN p.prior_week_start AND p.prior_week_end
                 THEN Date
             END) = 7
        THEN 'PASS'
        ELSE 'CHECK'
    END AS status

FROM social
CROSS JOIN params p

GROUP BY platform
ORDER BY platform;

SELECT
    view_source,
    MAX(Date) AS latest_date_available,
    COUNT(*) AS rows_on_latest_date,
    SUM(
        CASE
            WHEN Date = current_date() - INTERVAL 1 DAY
            THEN Spend
            ELSE 0
        END
    ) AS yesterday_spend
FROM prdrzranalytics.lab42.media_analytics_integrated_snapshot
WHERE Channel_Group_Name = 'Paid Social'
GROUP BY view_source
ORDER BY latest_date_available, view_source;
