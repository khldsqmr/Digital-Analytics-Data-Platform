-- 1. Full channel breadth + spend by channel/tactic in Display
SELECT Channel_Group_Name, Media_Tactic, Platform,
       COUNT(*) AS rows, SUM(Spend) AS total_spend,
       MIN(Date) AS min_date, MAX(Date) AS max_date
FROM prdrzranalytics.lab42.media_analytics_display
GROUP BY Channel_Group_Name, Media_Tactic, Platform
ORDER BY total_spend DESC;

-- 2. Full channel breadth + spend by channel/sub-channel in the integrated snapshot
SELECT Channel_Group_Name, Channel_Name,
       COUNT(*) AS rows, SUM(Spend) AS total_spend,
       MIN(Date) AS min_date, MAX(Date) AS max_date
FROM prdrzranalytics.lab42.media_analytics_integrated_snapshot
GROUP BY Channel_Group_Name, Channel_Name
ORDER BY total_spend DESC;

-- 3. What source views feed the integrated snapshot, and how much spend each contributes
SELECT view_source, Channel_Group_Name,
       COUNT(*) AS rows, SUM(Spend) AS total_spend
FROM prdrzranalytics.lab42.media_analytics_integrated_snapshot
GROUP BY view_source, Channel_Group_Name
ORDER BY view_source, total_spend DESC;

-- 4. Reconciliation: weekly spend for the 3 biddable channel groups in the integrated
-- snapshot, vs. the weekly totals already in your Biddable Spend Bronze table
SELECT Week_Start_Mon, Channel_Group_Name, SUM(Spend) AS integrated_spend
FROM prdrzranalytics.lab42.media_analytics_integrated_snapshot
WHERE Channel_Group_Name IN ('Programmatic', 'Paid Social', 'Paid Search')
GROUP BY Week_Start_Mon, Channel_Group_Name
ORDER BY Week_Start_Mon DESC, Channel_Group_Name;
