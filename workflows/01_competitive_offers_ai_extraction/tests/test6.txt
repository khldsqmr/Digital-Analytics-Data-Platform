SELECT
    lob,
    channel_group,
    SUM(metric_value) AS spend
FROM prdrzranalytics.lab42.sdi_tbl_dashboardPulseTms_silver_biddableSpend_weekly
WHERE metric_name = 'biddableSpend'
  AND qgp_date >= date_sub(current_date(), 35)
  AND lob IN ('POSTPAID', 'BROADBAND')
GROUP BY
    lob,
    channel_group
ORDER BY
    lob,
    channel_group;
