SELECT

    CASE

        WHEN alert_trigger = 1

        THEN CONCAT_WS(

            CHR(10),

            --------------------------------------------------------
            -- Run Information
            --------------------------------------------------------

            CONCAT(
                'Run Date: ',
                CAST(run_date AS STRING)
            ),

            CONCAT(
                'Tables Checked: ',
                CAST(total_tables AS STRING)
            ),

            CONCAT(
                'Successful Tables: ',
                CAST(successful_tables AS STRING)
            ),

            CONCAT(
                'Affected Tables: ',
                CAST(affected_table_count AS STRING)
            ),

            '',

            --------------------------------------------------------
            -- Failed Tables
            --------------------------------------------------------

            CASE
                WHEN failed_table_count > 0
                THEN CONCAT(
                    '🔴 Failed Tables (',
                    CAST(failed_table_count AS STRING),
                    '): ',
                    failed_tables
                )
            END,

            --------------------------------------------------------
            -- Skipped Tables
            --------------------------------------------------------

            CASE
                WHEN skipped_table_count > 0
                THEN CONCAT(
                    '🔴 Skipped Tables (',
                    CAST(skipped_table_count AS STRING),
                    '): ',
                    skipped_tables
                )
            END,

            --------------------------------------------------------
            -- Row-Count Mismatch
            --------------------------------------------------------

            CASE
                WHEN row_count_mismatch_count > 0
                THEN CONCAT(
                    '🟠 Row-Count Mismatch (',
                    CAST(row_count_mismatch_count AS STRING),
                    '): ',
                    row_count_mismatch_tables
                )
            END,

            --------------------------------------------------------
            -- Unknown Status
            --------------------------------------------------------

            CASE
                WHEN unknown_status_count > 0
                THEN CONCAT(
                    '🟠 Unknown Status (',
                    CAST(unknown_status_count AS STRING),
                    '): ',
                    unknown_status_tables
                )
            END,

            --------------------------------------------------------
            -- Schema Change
            --------------------------------------------------------

            CASE
                WHEN schema_change_count > 0
                THEN CONCAT(
                    '🟡 Schema Changes (',
                    CAST(schema_change_count AS STRING),
                    '): ',
                    schema_change_tables
                )
            END,

            '',

            --------------------------------------------------------
            -- Dashboard
            --------------------------------------------------------

            'Dashboard: https://adb-3157519905744188.8.azuredatabricks.net/dashboardsv3/01f1a173c1941a8781307551adc9950e/published?o=3157519905744188'

        )

        ELSE NULL

    END AS alert_message

FROM
    prdrzranalytics.lab42.sdi_vw_pipelineMonitoring_gold_bqUcAlertSummary_daily;
