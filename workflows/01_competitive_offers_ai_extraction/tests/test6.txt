-- 1. What kind of object is each source?
SELECT
    table_catalog,
    table_schema,
    table_name,
    table_type
FROM system.information_schema.tables
WHERE (table_catalog = 'prd_dbi_analytics'
       AND table_schema = 'improvado'
       AND table_name = 'pbi_programmatic_browsers_currentyr')
   OR (table_catalog = 'prdrzranalytics'
       AND table_schema = 'lab42'
       AND table_name IN (
           'media_analytics_integrated_snapshot',
           'sdi_tbl_sa360_gold_campaign_daily'
       ));


SHOW CREATE TABLE prd_dbi_analytics.improvado.pbi_programmatic_browsers_currentyr;

SHOW CREATE TABLE prdrzranalytics.lab42.sdi_tbl_sa360_gold_campaign_daily;

SELECT column_name
FROM prd_dbi_analytics.information_schema.columns
WHERE table_schema = 'improvado'
  AND table_name = 'pbi_programmatic_browsers_currentyr'
  AND (
       LOWER(column_name) LIKE '%account%'
    OR LOWER(column_name) LIKE '%advertiser%'
    OR LOWER(column_name) LIKE '%agency%'
    OR LOWER(column_name) LIKE '%dsp%'
    OR LOWER(column_name) LIKE '%lob%'
  )
ORDER BY column_name;

SELECT column_name
FROM prdrzranalytics.information_schema.columns
WHERE table_schema = 'lab42'
  AND table_name = 'media_analytics_integrated_snapshot'
  AND (
       LOWER(column_name) LIKE '%account%'
    OR LOWER(column_name) LIKE '%advertiser%'
    OR LOWER(column_name) LIKE '%agency%'
  )
ORDER BY column_name;

SHOW CREATE TABLE prdrzranalytics.lab42.sdi_tbl_sa360_gold_campaign_daily;

SELECT
    'Programmatic' AS source,
    MAX(date) AS latest_date,
    MIN(date) AS earliest_date
FROM prd_dbi_analytics.improvado.pbi_programmatic_browsers_currentyr

UNION ALL

SELECT
    'Paid Social',
    MAX(CAST(Date AS DATE)),
    MIN(CAST(Date AS DATE))
FROM prdrzranalytics.lab42.media_analytics_integrated_snapshot
WHERE Channel_Group_Name = 'Paid Social'
  AND Agency = 'InHouse'

UNION ALL

SELECT
    'Paid Search',
    MAX(date),
    MIN(date)
FROM prdrzranalytics.lab42.sdi_tbl_sa360_gold_campaign_daily;
