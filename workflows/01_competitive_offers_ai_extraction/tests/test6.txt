SELECT
    TRY_CAST(job_run_ts AS TIMESTAMP) AS job_run_ts,
    CAST(TRY_CAST(job_run_ts AS TIMESTAMP) AS DATE) AS run_date,
    bq_table,
    status,
    cols_added,
    cols_removed,
    cols_changed,
    CASE
        WHEN cols_added IS NOT NULL
          OR cols_removed IS NOT NULL
          OR cols_changed IS NOT NULL
        THEN 'Schema Change'
        ELSE 'No Schema Change'
    END AS schema_change_status
FROM prd_dbi_analytics.improvado.bq_uc_pipeline_run_log
WHERE LOWER(bq_table) LIKE '%mrt%social%'
ORDER BY TRY_CAST(job_run_ts AS TIMESTAMP) DESC;
