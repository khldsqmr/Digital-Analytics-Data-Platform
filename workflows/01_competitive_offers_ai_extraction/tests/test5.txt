  MAX(week_ending) AS latest_week_with_value

SELECT
    'Activation' AS MetricName,
    WeekEnding,
    MetricType,
    SUM(Amount) AS Amount
FROM prdrzrlakehouse.qgp_restricted.qgpweeklyview
WHERE MetricType IN ('QGP', 'Actuals', 'Actuals/Outlook')
  AND
   upper(trim(MetricID)) IN (
        upper(trim('ConsumerPostpaidNewPhoneBANBOPISUnassistedActivationsTM1MappedDigital')),
        upper(trim('ConsumerPostpaidNewPhoneBANNonBOPISUnassistedActivationsTM1MappedDigital'))
      )
  AND DateContext = 'Normal'
  AND Page = 'Digital Transformation NEW'
GROUP BY
    WeekEnding,
    MetricType
ORDER BY WeekEnding desc;
