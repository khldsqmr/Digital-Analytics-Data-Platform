-- ============================================================
-- UNIQUE CATEGORIES FOR ALL DIMENSION COLUMNS
--
-- Dimension definition:
--   STRING + BOOLEAN columns
--
-- Scope:
--   Programmatic = full source universe
--   Paid Social  = Channel_Group_Name = 'Paid Social'
--                  (all agencies / LOBs retained)
--   Paid Search  = full source universe
--
-- Output is long-form and Excel-friendly.
-- ============================================================

DECLARE OR REPLACE VARIABLE category_sql STRING;

SET VAR category_sql = (

    WITH source_columns AS (

        -- ====================================================
        -- 1. PROGRAMMATIC
        -- ====================================================
        SELECT
            1 AS source_order,
            'Programmatic' AS source_channel,
            'pbi_programmatic_browsers_currentyr' AS source_table,
            'prd_dbi_analytics.improvado.pbi_programmatic_browsers_currentyr'
                AS full_table_name,
            '1 = 1' AS row_filter,

            ordinal_position,
            column_name,
            data_type

        FROM prd_dbi_analytics.information_schema.columns

        WHERE table_schema = 'improvado'
          AND table_name = 'pbi_programmatic_browsers_currentyr'
          AND UPPER(data_type) IN ('STRING', 'BOOLEAN')


        UNION ALL


        -- ====================================================
        -- 2. PAID SOCIAL
        -- ====================================================
        SELECT
            2 AS source_order,
            'Paid Social' AS source_channel,
            'media_analytics_integrated_snapshot' AS source_table,
            'prdrzranalytics.lab42.media_analytics_integrated_snapshot'
                AS full_table_name,

            -- Universe for Paid Social only.
            -- Do NOT restrict Agency or LOB here.
            'Channel_Group_Name = ''Paid Social''' AS row_filter,

            ordinal_position,
            column_name,
            data_type

        FROM prdrzranalytics.information_schema.columns

        WHERE table_schema = 'lab42'
          AND table_name = 'media_analytics_integrated_snapshot'
          AND UPPER(data_type) IN ('STRING', 'BOOLEAN')


        UNION ALL


        -- ====================================================
        -- 3. PAID SEARCH
        -- ====================================================
        SELECT
            3 AS source_order,
            'Paid Search' AS source_channel,
            'sdi_tbl_sa360_gold_campaign_daily' AS source_table,
            'prdrzranalytics.lab42.sdi_tbl_sa360_gold_campaign_daily'
                AS full_table_name,
            '1 = 1' AS row_filter,

            ordinal_position,
            column_name,
            data_type

        FROM prdrzranalytics.information_schema.columns

        WHERE table_schema = 'lab42'
          AND table_name = 'sdi_tbl_sa360_gold_campaign_daily'
          AND UPPER(data_type) IN ('STRING', 'BOOLEAN')
    ),


    -- Build the STACK arguments dynamically.
    --
    -- Each dimension contributes:
    --   ordinal_position,
    --   dimension_name,
    --   data_type,
    --   value
    dimension_expressions AS (

        SELECT
            source_order,
            source_channel,
            source_table,
            full_table_name,
            row_filter,
            ordinal_position,

            CONCAT(
                CAST(ordinal_position AS STRING),
                ', ',
                '''',
                REPLACE(column_name, '''', ''''''),
                ''', ',
                '''',
                REPLACE(data_type, '''', ''''''),
                ''', ',
                'CAST(`',
                REPLACE(column_name, '`', '``'),
                '` AS STRING)'
            ) AS stack_expression

        FROM source_columns
    ),


    -- Build one query per source table.
    source_queries AS (

        SELECT
            source_order,

            CONCAT(
                'SELECT ',
                    '''', source_channel, ''' AS source_channel, ',
                    '''', source_table, ''' AS source_table, ',
                    'dimension_order, ',
                    'dimension_name, ',
                    'data_type, ',
                    'COALESCE(category_value, ''[NULL]'') AS category_value, ',
                    'COUNT(*) AS row_count ',

                'FROM ',
                    full_table_name, ' ',

                'LATERAL VIEW STACK(',
                    CAST(COUNT(*) AS STRING),
                    ', ',

                    ARRAY_JOIN(
                        TRANSFORM(
                            SORT_ARRAY(
                                COLLECT_LIST(
                                    NAMED_STRUCT(
                                        'ord', ordinal_position,
                                        'expr', stack_expression
                                    )
                                )
                            ),
                            x -> x.expr
                        ),
                        ', '
                    ),

                ') category_stack ',
                    'AS dimension_order, dimension_name, data_type, category_value ',

                'WHERE ',
                    row_filter, ' ',

                'GROUP BY ',
                    'dimension_order, ',
                    'dimension_name, ',
                    'data_type, ',
                    'category_value'
            ) AS source_sql

        FROM dimension_expressions

        GROUP BY
            source_order,
            source_channel,
            source_table,
            full_table_name,
            row_filter
    )


    -- Combine the three generated queries.
    SELECT
        CONCAT(
            'SELECT * FROM (',

            ARRAY_JOIN(
                TRANSFORM(
                    SORT_ARRAY(
                        COLLECT_LIST(
                            NAMED_STRUCT(
                                'ord', source_order,
                                'sql', source_sql
                            )
                        )
                    ),
                    x -> x.sql
                ),
                ' UNION ALL '
            ),

            ') categories ',
            'ORDER BY ',
                'source_channel, ',
                'dimension_order, ',
                'category_value'
        )

    FROM source_queries
);


-- Run dynamically generated query.
EXECUTE IMMEDIATE category_sql;
