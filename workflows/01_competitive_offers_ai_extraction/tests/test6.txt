-- PAID SOCIAL READINESS
-- Week being tested: Sun Sep 6 - Sat Sep 12
-- Checkpoints: Mon/Tue/Wed 10 AM ET + Thu 6 AM ET

WITH mon AS (
    SELECT
        'Mon 10 AM ET' AS checkpoint,
        REPLACE(TRIM(Channel_Name), 'Paid Social - ', '') AS platform,
        COUNT(DISTINCT CAST(Date AS DATE)) AS days_present,
        SUM(TRY_CAST(Spend AS DOUBLE)) AS checkpoint_spend
    FROM prdrzranalytics.lab42.media_analytics_integrated_snapshot
    TIMESTAMP AS OF '2026-09-14 14:00:00'
    WHERE Channel_Group_Name = 'Paid Social'
      AND Agency = 'InHouse'
      AND UPPER(TRIM(LOB)) IN ('POSTPAID','BROADBAND','HSI')
      AND CAST(Date AS DATE) BETWEEN '2026-09-06' AND '2026-09-12'
    GROUP BY REPLACE(TRIM(Channel_Name), 'Paid Social - ', '')
),

tue AS (
    SELECT
        'Tue 10 AM ET' AS checkpoint,
        REPLACE(TRIM(Channel_Name), 'Paid Social - ', '') AS platform,
        COUNT(DISTINCT CAST(Date AS DATE)) AS days_present,
        SUM(TRY_CAST(Spend AS DOUBLE)) AS checkpoint_spend
    FROM prdrzranalytics.lab42.media_analytics_integrated_snapshot
    TIMESTAMP AS OF '2026-09-15 14:00:00'
    WHERE Channel_Group_Name = 'Paid Social'
      AND Agency = 'InHouse'
      AND UPPER(TRIM(LOB)) IN ('POSTPAID','BROADBAND','HSI')
      AND CAST(Date AS DATE) BETWEEN '2026-09-06' AND '2026-09-12'
    GROUP BY REPLACE(TRIM(Channel_Name), 'Paid Social - ', '')
),

wed AS (
    SELECT
        'Wed 10 AM ET' AS checkpoint,
        REPLACE(TRIM(Channel_Name), 'Paid Social - ', '') AS platform,
        COUNT(DISTINCT CAST(Date AS DATE)) AS days_present,
        SUM(TRY_CAST(Spend AS DOUBLE)) AS checkpoint_spend
    FROM prdrzranalytics.lab42.media_analytics_integrated_snapshot
    TIMESTAMP AS OF '2026-09-16 14:00:00'
    WHERE Channel_Group_Name = 'Paid Social'
      AND Agency = 'InHouse'
      AND UPPER(TRIM(LOB)) IN ('POSTPAID','BROADBAND','HSI')
      AND CAST(Date AS DATE) BETWEEN '2026-09-06' AND '2026-09-12'
    GROUP BY REPLACE(TRIM(Channel_Name), 'Paid Social - ', '')
),

thu AS (
    SELECT
        'Thu 6 AM ET' AS checkpoint,
        REPLACE(TRIM(Channel_Name), 'Paid Social - ', '') AS platform,
        COUNT(DISTINCT CAST(Date AS DATE)) AS days_present,
        SUM(TRY_CAST(Spend AS DOUBLE)) AS checkpoint_spend
    FROM prdrzranalytics.lab42.media_analytics_integrated_snapshot
    TIMESTAMP AS OF '2026-09-17 10:00:00'
    WHERE Channel_Group_Name = 'Paid Social'
      AND Agency = 'InHouse'
      AND UPPER(TRIM(LOB)) IN ('POSTPAID','BROADBAND','HSI')
      AND CAST(Date AS DATE) BETWEEN '2026-09-06' AND '2026-09-12'
    GROUP BY REPLACE(TRIM(Channel_Name), 'Paid Social - ', '')
),

checkpoints AS (
    SELECT * FROM mon
    UNION ALL
    SELECT * FROM tue
    UNION ALL
    SELECT * FROM wed
    UNION ALL
    SELECT * FROM thu
),

final_spend AS (
    SELECT
        REPLACE(TRIM(Channel_Name), 'Paid Social - ', '') AS platform,
        SUM(TRY_CAST(Spend AS DOUBLE)) AS final_spend
    FROM prdrzranalytics.lab42.media_analytics_integrated_snapshot
    WHERE Channel_Group_Name = 'Paid Social'
      AND Agency = 'InHouse'
      AND UPPER(TRIM(LOB)) IN ('POSTPAID','BROADBAND','HSI')
      AND CAST(Date AS DATE) BETWEEN '2026-09-06' AND '2026-09-12'
    GROUP BY REPLACE(TRIM(Channel_Name), 'Paid Social - ', '')
)

SELECT
    c.checkpoint,
    c.platform,
    c.days_present,
    c.checkpoint_spend,
    f.final_spend,
    ROUND(
        100 * c.checkpoint_spend / NULLIF(f.final_spend, 0),
        1
    ) AS pct_of_final,
    CASE
        WHEN c.days_present = 7
         AND c.checkpoint_spend / NULLIF(f.final_spend, 0) >= 0.99
        THEN 'READY'
        ELSE 'NOT READY'
    END AS readiness
FROM checkpoints c
LEFT JOIN final_spend f
    ON c.platform = f.platform
ORDER BY
    CASE c.checkpoint
        WHEN 'Mon 10 AM ET' THEN 1
        WHEN 'Tue 10 AM ET' THEN 2
        WHEN 'Wed 10 AM ET' THEN 3
        WHEN 'Thu 6 AM ET' THEN 4
    END,
    c.platform;
