  Expect: every week from 2026-09-05 at about 100 percent; the newest partial week may be lower only if CM360 lags DV360. */

  AND campaign_name = 'TMO_PostpaidHINT_VoiceHINTFiber_X_PSTPHINTPro_Perf_VINT_X_EL_DIS_2026_Multi-Opt_PR-OEX_DV360_X_US_DSP-CMP-001'

/* Q2  BRONZE: the latest run copied the source exactly (no rows lost or multiplied)
  Expect: source_rows = bronze_rows and source_spend = bronze_spend. */
SELECT s.source_rows, b.bronze_rows, s.source_spend, b.bronze_spend
FROM (
  SELECT COUNT(*) AS source_rows, ROUND(SUM(TRY_CAST(spend AS DOUBLE)), 2) AS source_spend
  FROM prd_dbi_analytics.improvado.pbi_programmatic_browsers_currentyr
  WHERE CAST(`date` AS DATE) IS NOT NULL
) s
CROSS JOIN (
  SELECT COUNT(*) AS bronze_rows, ROUND(SUM(TRY_CAST(spend AS DOUBLE)), 2) AS bronze_spend
  FROM prdrzranalytics.lab42.sdi_tbl_biddProgrammaticArchive_bronze_retained_weekly
  WHERE archive_run_id = (SELECT archive_run_id FROM prdrzranalytics.lab42.sdi_tbl_biddProgrammaticArchive_bronze_retained_weekly ORDER BY archive_loaded_ts DESC LIMIT 1)
) b;
/*
