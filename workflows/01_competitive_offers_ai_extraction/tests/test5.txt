SELECT
    run_date,
    alert_trigger,
    monitoring_status,
    total_tables,
    successful_tables,
    affected_table_count,
    failed_table_count,
    unknown_status_count,
    row_count_mismatch_count,
    schema_change_count,
    skipped_table_count,
    issue_summary,
    affected_tables,
    monitoring_dashboard_url,

    CASE
        WHEN alert_trigger = 1
        THEN CONCAT(
            monitoring_status,
            ' | Run Date: ', CAST(run_date AS STRING),
            ' | Affected Tables: ', CAST(affected_table_count AS STRING),
            ' | Issues: ', COALESCE(issue_summary, 'N/A'),
            ' | Tables: ', COALESCE(affected_tables, 'N/A')
        )
        ELSE NULL
    END AS alert_message

FROM
    prdrzranalytics.lab42.sdi_vw_pipelineMonitoring_gold_bqUcAlertSummary_daily;
