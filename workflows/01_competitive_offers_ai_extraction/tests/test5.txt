SELECT
  WeekEnding, MetricID, MetricType, Page, DisplayMetricType, IsFuture, Amount, InsertDateTime, PublishKey,
  ROW_NUMBER() OVER (
    PARTITION BY WeekEnding, MetricID, DateContext, MetricType, Page
    ORDER BY InsertDateTime DESC, PublishKey DESC
  ) AS rn
FROM prdrzrlakehouse.qgp_restricted.qgpweeklyview
WHERE UPPER(TRIM(MetricID)) = UPPER('VRInboundCallsinclHSIAutomatedManual')
  AND UPPER(TRIM(MetricType)) = 'QGP'
  AND UPPER(TRIM(DisplayMetricType)) = 'TARGET'
  AND UPPER(TRIM(DateContext)) = 'NORMAL'
  AND UPPER(TRIM(Page)) = 'VIRTUAL RETAIL'
ORDER BY WeekEnding DESC, InsertDateTime DESC;
