// c.tms.qgp.Primary Metric
CASE [p.qgp.Primary Metric QGP]
  WHEN 'Store Traffic Actuals'         THEN [c.tms.Exit Traffic Actuals]
  WHEN 'Store Traffic QGP'             THEN [c.tms.Exit Traffic QGP]
  WHEN 'VR Calls Actuals'              THEN [c.tms.VR Calls Actuals]
  WHEN 'VR Calls QGP'                  THEN [c.tms.VR Calls QGP]
  WHEN 'VR Chats Actuals'              THEN [c.tms.VR Chats Actuals]
  WHEN 'VR Chats QGP'                  THEN [c.tms.VR Chats QGP]
  WHEN 'VR New BANs Actuals'           THEN [c.tms.VR New BANs Actuals]
  WHEN 'VR New BANs QGP'               THEN [c.tms.VR New BANs QGP]
  WHEN 'Activations BOPIS Actuals'     THEN [c.tms.Activations BOPIS Actuals]
  WHEN 'Activations BOPIS QGP'         THEN [c.tms.Activations BOPIS QGP]
  WHEN 'MFC Spend Actual (Postpaid)'   THEN SUM(IF [Data Source] = 'MFC_SPEND_CHANNEL' AND [Metric Name] = 'mfcSpendActual' AND [Lob] = 'POSTPAID'   AND [Channel Group] = 'All Channels' THEN [Metric Value] END)
  WHEN 'MFC Spend Actual (Broadband)'  THEN SUM(IF [Data Source] = 'MFC_SPEND_CHANNEL' AND [Metric Name] = 'mfcSpendActual' AND [Lob] = 'BROADBAND'  AND [Channel Group] = 'All Channels' THEN [Metric Value] END)
  WHEN 'MFC Spend Actual (Total)'      THEN SUM(IF [Data Source] = 'MFC_SPEND_CHANNEL' AND [Metric Name] = 'mfcSpendActual' AND [Lob] IN ('POSTPAID', 'BROADBAND') AND [Channel Group] = 'All Channels' THEN [Metric Value] END)
  WHEN 'MFC Spend Forecast (Postpaid)' THEN SUM(IF [Data Source] = 'MFC_SPEND_CHANNEL' AND [Metric Name] = 'mfcSpendForecast' AND [Lob] = 'POSTPAID'  AND [Channel Group] = 'All Channels' THEN [Metric Value] END)
  WHEN 'MFC Spend Forecast (Broadband)' THEN SUM(IF [Data Source] = 'MFC_SPEND_CHANNEL' AND [Metric Name] = 'mfcSpendForecast' AND [Lob] = 'BROADBAND' AND [Channel Group] = 'All Channels' THEN [Metric Value] END)
  WHEN 'MFC Spend Forecast (Total)'    THEN SUM(IF [Data Source] = 'MFC_SPEND_CHANNEL' AND [Metric Name] = 'mfcSpendForecast' AND [Lob] IN ('POSTPAID', 'BROADBAND') AND [Channel Group] = 'All Channels' THEN [Metric Value] END)
  WHEN 'Platform Spend (Postpaid)'     THEN SUM(IF [Data Source] = 'PLATFORM_SPEND_CHANNEL' AND [Metric Name] = 'platformSpend' AND [Lob] = 'POSTPAID'   AND [Channel Group] = 'All Channels' THEN [Metric Value] END)
  WHEN 'Platform Spend (Broadband)'    THEN SUM(IF [Data Source] = 'PLATFORM_SPEND_CHANNEL' AND [Metric Name] = 'platformSpend' AND [Lob] = 'BROADBAND'  AND [Channel Group] = 'All Channels' THEN [Metric Value] END)
  WHEN 'Platform Spend (Total)'        THEN SUM(IF [Data Source] = 'PLATFORM_SPEND_CHANNEL' AND [Metric Name] = 'platformSpend' AND [Lob] IN ('POSTPAID', 'BROADBAND') AND [Channel Group] = 'All Channels' THEN [Metric Value] END)
  ELSE NULL
END
