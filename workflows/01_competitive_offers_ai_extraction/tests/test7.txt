SELECT source_table, page_app_type, attribute_event_type, COUNT(*) AS n
FROM prd_dbi_analytics.silver_digital_interactions.unified_digital_interactions
WHERE event_date = DATE '2026-09-23'
GROUP BY ALL
ORDER BY n DESC;

SELECT source_datalakeLoadDate, COUNT(*) AS n
FROM prd_dbi_analytics.silver_digital_interactions.unified_digital_interactions
WHERE event_date = DATE '2026-09-20'
GROUP BY 1
ORDER BY 1;

SELECT
  SUM(CASE WHEN try_cast(event_collection_timestamp_utc AS TIMESTAMP) IS NULL
            AND event_collection_timestamp_utc IS NOT NULL THEN 1 ELSE 0 END) AS bad_collection_ts,
  SUM(CASE WHEN length(participant_key) = 31 THEN 1 ELSE 0 END) AS pk_31_chars,
  SUM(CASE WHEN participant_key LIKE 'device_id_%' THEN 1 ELSE 0 END) AS pk_device_id,
  SUM(CASE WHEN hit_id <> lower(hit_id) THEN 1 ELSE 0 END) AS upper_hit_ids
FROM prd_dbi_analytics.silver_digital_interactions.unified_digital_interactions
WHERE event_date = DATE '2026-09-23';
