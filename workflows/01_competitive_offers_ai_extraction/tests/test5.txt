SELECT metric_id, metric_type, date_context, page, display_metric_type, is_future, COUNT(*) AS n, MAX(week_ending) AS most_recent_week
FROM prdrzranalytics.lab42.sdi_tbl_dashboardPulseTms_bronze_qgp_weekly
WHERE UPPER(TRIM(metric_id)) IN (
  'CONSUMERPOSTPAIDNEWPHONEBANBOPISUNASSISTEDACTIVATIONSTM1MAPPEDDIGITAL',
  'CONSUMERPOSTPAIDNEWPHONEBANNONBOPISUNASSISTEDACTIVATIONSTM1MAPPEDDIGITAL',
  'CONSUMERPOSTPAIDNEWPHONEBANACTIVATIONSTM1MAPPEDDIGITAL',
  'DIGITALPCTOFCONSUMERPOSTPAIDACTIVATIONSTOTALINCLASSISTEDNEW',
  'VRINBOUNDCALLSINCLHSIAUTOMATEDMANUAL',
  'VRCHATSINCLHSIAUTOMATEDMANUAL'
)
GROUP BY metric_id, metric_type, date_context, page, display_metric_type, is_future
ORDER BY metric_id, metric_type;
