SELECT MetricID, WeekEnding, MetricType, Page, DateContext, DisplayMetricType, Amount
FROM prdrzrlakehouse.qgp_restricted.qgpweeklyview
WHERE UPPER(TRIM(MetricID)) = 'DIGITALPCTOFCONSUMERPOSTPAIDACTIVATIONSEXCLFIBERNRINDIRECTCHANNELBTSODA'
  AND UPPER(TRIM(DateContext)) = 'NORMAL'
ORDER BY WeekEnding DESC
LIMIT 20;

SELECT DISTINCT MetricID, MetricName, Page
FROM prdrzrlakehouse.qgp_restricted.qgpweeklyview
WHERE LOWER(MetricName) LIKE '%aal%' OR LOWER(MetricID) LIKE '%aal%'
ORDER BY MetricName, MetricID;
