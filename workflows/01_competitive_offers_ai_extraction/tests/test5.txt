SELECT
  SAFE_CAST(NULLIF(CAST(QGP_Week AS STRING), 'None') AS DATE) AS QGP_Week,
  SUM(Spend) AS home_internet_spend,
  COUNT(*) AS source_row_count
FROM
  `prj-dbi-prd-1.ds_dbi_marketing.ma_mfc_raw`
WHERE
  UPPER(TRIM(LOB_Supported)) IN ('BROADBAND', 'HSI')
  AND UPPER(TRIM(WM_NWM)) = 'WORKING'
  AND UPPER(TRIM(QGP)) = 'ACTUAL'
  AND Channel IS NOT NULL
  AND Channel NOT IN ('OTHER (do not use)', 'Non-Working', 'Budget Held')
  AND Spend IS NOT NULL
GROUP BY
  QGP_Week
ORDER BY
  QGP_Week;
