SELECT PublishKey, COUNT(*) AS n_rows, COUNT(DISTINCT WeekEnding) AS n_weeks, MIN(InsertDateTime) AS min_insert, MAX(InsertDateTime) AS max_insert
FROM prdrzrlakehouse.qgp_restricted.qgpweeklyview
GROUP BY PublishKey
ORDER BY PublishKey DESC
LIMIT 20;

SELECT MetricID, WeekEnding, MetricType, Page, COUNT(DISTINCT PublishKey) AS n_publish_keys, COUNT(DISTINCT Amount) AS n_distinct_amounts, MIN(Amount) AS min_amt, MAX(Amount) AS max_amt
FROM prdrzrlakehouse.qgp_restricted.qgpweeklyview
WHERE UPPER(TRIM(MetricID)) IN (
  'CONSUMERPOSTPAIDNEWPHONEBANBOPISUNASSISTEDACTIVATIONSTM1MAPPEDDIGITAL',
  'CONSUMERPOSTPAIDNEWPHONEBANNONBOPISUNASSISTEDACTIVATIONSTM1MAPPEDDIGITAL',
  'CONSUMERPOSTPAIDNEWPHONEBANACTIVATIONSTM1MAPPEDDIGITAL',
  'VRINBOUNDCALLSINCLHSIAUTOMATEDMANUAL',
  'VRCHATSINCLHSIAUTOMATEDMANUAL',
  'VRPOSTPAIDACTIVATIONSINCLVIRTUALBUSINESSTM1MAPPED',
  'CONPOSTPAIDDIGITALPCTOFNEWBANPHONEUNASSISTEDANDASSISTEDWONRINDIRECT',
  'DIGITALPCTOFCONSUMERPOSTPAIDACTIVATIONSTOTALINCLASSISTEDNEW',
  'DIGITALPCTOFCONSUMERPOSTPAIDACTIVATIONSPHONENEWSAMEPAGEWONRINDIRECT',
  'NEWPHONEBANASSISTEDACTIVATIONSDIGITALPCTOFCONSUMERPOSTPAIDTM1MAPPEDWONRINDIRECT'
)
AND UPPER(TRIM(DateContext)) = 'NORMAL'
GROUP BY MetricID, WeekEnding, MetricType, Page
HAVING COUNT(DISTINCT PublishKey) > 1
ORDER BY n_distinct_amounts DESC, MetricID, WeekEnding
LIMIT 50;
