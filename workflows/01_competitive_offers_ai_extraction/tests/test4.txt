SELECT
  r.Quarter AS Raw_Quarter,
  CONCAT('20', SUBSTRING(r.Quarter, 4, 2), ' ', SUBSTRING(r.Quarter, 1, 2)) AS Normalized_Quarter,
  s.QGP_Week, s.Quarter AS Silver_Quarter, s.week_type, s.Spend_Forecast
FROM prdrzranalytics.lab42.raw_media_flowchart r
JOIN prdrzranalytics.lab42.sdi_tbl_mfc_silver_spendGranular_weekly s
  ON DATE(r.Week_Beginning_Monday) = s.Week_Beginning_Monday
 AND UPPER(TRIM(r.LOB_Supported)) = s.LOB_Supported
 AND r.Channel <=> s.Channel AND r.Tactic <=> s.Tactic
 AND r.Message_Type <=> s.Message_Type
 AND r.Agency <=> s.Agency
WHERE UPPER(TRIM(r.QGP)) = 'FORECAST'
  AND r.Week_Beginning_Monday = '2026-03-30'
  AND UPPER(TRIM(r.LOB_Supported)) = 'CONSUMER POSTPAID'
ORDER BY r.Channel, r.Tactic, r.Message_Type, r.Agency, r.Quarter;
