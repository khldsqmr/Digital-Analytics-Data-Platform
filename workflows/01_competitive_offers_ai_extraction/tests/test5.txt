c.Store Traffic Actual
= SUM(IF [metric_name] = "storeTraffic" AND [metric_type] = "QGP_ACTUAL" THEN [metric_value] END)

c.Store Traffic Target
= SUM(IF [metric_name] = "storeTraffic" AND [metric_type] = "QGP_TARGET" THEN [metric_value] END)


c.Store Traffic AvT %
= ([c.Store Traffic Actual] - [c.Store Traffic Target]) / ABS([c.Store Traffic Target])

c.Store Traffic AvT % (Pos)
= IF [c.Store Traffic AvT %] > 0 THEN [c.Store Traffic AvT %] END

c.Store Traffic AvT % (Neg)
= IF [c.Store Traffic AvT %] < 0 THEN [c.Store Traffic AvT %] END

c.UPV WoW %
= ATTR(IF [metric_name] = "upvTotalAdobe" AND [data_source] = "ADOBE" THEN [wow_pct] END)

c.UPV WoW % (Pos)
= IF [c.UPV WoW %] > 0 THEN [c.UPV WoW %] END

c.UPV WoW % (Neg)
= IF [c.UPV WoW %] < 0 THEN [c.UPV WoW %] END
