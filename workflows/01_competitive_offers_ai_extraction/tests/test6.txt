SELECT
  lob,
  channel_group,
  COUNT(*) AS row_count,
  SUM(spend) AS spend
FROM prdrzranalytics.lab42.sdi_tbl_dashboardPulseTms_bronze_biddableSpend_weekly
WHERE week_sun_sat >= date_sub(current_date(), 35)
  AND lob IN ('POSTPAID', 'HSI', 'BROADBAND')
GROUP BY
  lob,
  channel_group
ORDER BY
  lob,
  channel_group;
