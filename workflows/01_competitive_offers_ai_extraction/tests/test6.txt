WITH base AS (
    SELECT
        CAST(date AS DATE) AS date,
        TRY_CAST(spend AS DOUBLE) AS spend,
        NULLIF(TRIM(lob),'') AS lob,
        NULLIF(TRIM(DSP),'') AS dsp,
        NULLIF(TRIM(campaign_type),'') AS campaign_type,
        NULLIF(TRIM(business_initiative),'') AS business_initiative,
        NULLIF(TRIM(budget_source),'') AS budget_source,
        NULLIF(TRIM(budget_product_line),'') AS budget_product_line,
        NULLIF(TRIM(program),'') AS program,
        NULLIF(TRIM(creative_type),'') AS creative_type
    FROM prd_dbi_analytics.improvado.pbi_programmatic_browsers_currentyr
)

SELECT 'LOB' AS dimension, COALESCE(lob,'<NULL>') AS category,
       COUNT(*) AS row_count, ROUND(SUM(spend),2) AS spend,
       MIN(date) AS min_date, MAX(date) AS max_date
FROM base
GROUP BY COALESCE(lob,'<NULL>')

UNION ALL

SELECT 'DSP / Platform', COALESCE(dsp,'<NULL>'),
       COUNT(*), ROUND(SUM(spend),2), MIN(date), MAX(date)
FROM base
GROUP BY COALESCE(dsp,'<NULL>')

UNION ALL

SELECT 'Campaign Type', COALESCE(campaign_type,'<NULL>'),
       COUNT(*), ROUND(SUM(spend),2), MIN(date), MAX(date)
FROM base
GROUP BY COALESCE(campaign_type,'<NULL>')

UNION ALL

SELECT 'Business Initiative', COALESCE(business_initiative,'<NULL>'),
       COUNT(*), ROUND(SUM(spend),2), MIN(date), MAX(date)
FROM base
GROUP BY COALESCE(business_initiative,'<NULL>')

UNION ALL

SELECT 'Budget Source', COALESCE(budget_source,'<NULL>'),
       COUNT(*), ROUND(SUM(spend),2), MIN(date), MAX(date)
FROM base
GROUP BY COALESCE(budget_source,'<NULL>')

UNION ALL

SELECT 'Budget Product Line', COALESCE(budget_product_line,'<NULL>'),
       COUNT(*), ROUND(SUM(spend),2), MIN(date), MAX(date)
FROM base
GROUP BY COALESCE(budget_product_line,'<NULL>')

UNION ALL

SELECT 'Program', COALESCE(program,'<NULL>'),
       COUNT(*), ROUND(SUM(spend),2), MIN(date), MAX(date)
FROM base
GROUP BY COALESCE(program,'<NULL>')

UNION ALL

SELECT 'Creative Type', COALESCE(creative_type,'<NULL>'),
       COUNT(*), ROUND(SUM(spend),2), MIN(date), MAX(date)
FROM base
GROUP BY COALESCE(creative_type,'<NULL>')

ORDER BY dimension, spend DESC;

    UPPER(TRIM(ad_platform)) AS ad_platform,
    COALESCE(campaign_type,'<NULL>') AS campaign_type,
    COALESCE(advertising_channel_type,'<NULL>') AS advertising_channel_type,
    COUNT(DISTINCT account_id) AS accounts,
    COUNT(DISTINCT campaign_id) AS campaigns,

    MIN(date) AS min_date,
    MAX(date) AS max_date

    'GOOGLE','GOOGLE ADS','GOOGLE_ADS',
    'BING','MICROSOFT','MICROSOFT ADS','MICROSOFT_ADS'
)
AND UPPER(TRIM(lob)) IN (
    'POSTPAID','HSI','BROADBAND','FIBER'
)

    UPPER(TRIM(lob)),
    UPPER(TRIM(ad_platform)),
    COALESCE(campaign_type,'<NULL>'),
    COALESCE(advertising_channel_type,'<NULL>')
ORDER BY lob, ad_platform, spend DESC;
