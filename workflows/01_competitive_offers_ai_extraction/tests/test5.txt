CREATE OR REPLACE VIEW
prdrzranalytics.lab42.sdi_vw_pipelineMonitoring_gold_bqUcAlertSummary_daily
AS

WITH latest_run AS (

    -- Get the latest available BQ → UC pipeline run date
    SELECT
        run_date
    FROM
        prdrzranalytics.lab42.sdi_vw_pipelineMonitoring_gold_bqUcRunDetails_daily
    WHERE
        run_date IS NOT NULL
    ORDER BY
        run_date DESC,
        job_run_ts DESC NULLS LAST,
        job_end_ts DESC NULLS LAST
    LIMIT 1

),

health AS (

    -- Get the health summary for the latest run
    SELECT
        h.run_date,
        MAX(h.total_tables)      AS total_tables,
        MAX(h.successful_tables) AS successful_tables

    FROM
        prdrzranalytics.lab42.sdi_vw_pipelineMonitoring_gold_bqUcHealthSummary_daily h

    INNER JOIN latest_run l
        ON h.run_date = l.run_date

    GROUP BY
        h.run_date

),

issues AS (

    -- Get all table-level issues for the latest run
    SELECT
        i.*
    FROM
        prdrzranalytics.lab42.sdi_vw_pipelineMonitoring_gold_bqUcTableIssues_daily i

    INNER JOIN latest_run l
        ON i.run_date = l.run_date

),

issue_rollup AS (

    SELECT

        COUNT(*) AS issue_record_count,

        COUNT(DISTINCT bq_table)
            AS affected_table_count,

        ------------------------------------------------------------
        -- ISSUE COUNTS
        ------------------------------------------------------------

        COUNT(
            DISTINCT CASE
                WHEN issue_type = 'Failed Table'
                THEN bq_table
            END
        ) AS failed_table_count,

        COUNT(
            DISTINCT CASE
                WHEN issue_type = 'Skipped Table'
                THEN bq_table
            END
        ) AS skipped_table_count,

        COUNT(
            DISTINCT CASE
                WHEN issue_type = 'Unknown Status'
                THEN bq_table
            END
        ) AS unknown_status_count,

        COUNT(
            DISTINCT CASE
                WHEN issue_type = 'Row-Count Mismatch'
                THEN bq_table
            END
        ) AS row_count_mismatch_count,

        COUNT(
            DISTINCT CASE
                WHEN issue_type = 'Schema Change'
                THEN bq_table
            END
        ) AS schema_change_count,

        ------------------------------------------------------------
        -- TABLE LISTS BY ISSUE TYPE
        ------------------------------------------------------------

        CONCAT_WS(
            ', ',
            SORT_ARRAY(
                COLLECT_SET(
                    CASE
                        WHEN issue_type = 'Failed Table'
                        THEN bq_table
                    END
                )
            )
        ) AS failed_tables,

        CONCAT_WS(
            ', ',
            SORT_ARRAY(
                COLLECT_SET(
                    CASE
                        WHEN issue_type = 'Skipped Table'
                        THEN bq_table
                    END
                )
            )
        ) AS skipped_tables,

        CONCAT_WS(
            ', ',
            SORT_ARRAY(
                COLLECT_SET(
                    CASE
                        WHEN issue_type = 'Unknown Status'
                        THEN bq_table
                    END
                )
            )
        ) AS unknown_status_tables,

        CONCAT_WS(
            ', ',
            SORT_ARRAY(
                COLLECT_SET(
                    CASE
                        WHEN issue_type = 'Row-Count Mismatch'
                        THEN bq_table
                    END
                )
            )
        ) AS row_count_mismatch_tables,

        CONCAT_WS(
            ', ',
            SORT_ARRAY(
                COLLECT_SET(
                    CASE
                        WHEN issue_type = 'Schema Change'
                        THEN bq_table
                    END
                )
            )
        ) AS schema_change_tables

    FROM issues

),

base AS (

    SELECT

        l.run_date,

        ------------------------------------------------------------
        -- ALERT TRIGGER
        ------------------------------------------------------------

        CASE
            WHEN COALESCE(i.issue_record_count, 0) > 0
            THEN 1
            ELSE 0
        END AS alert_trigger,

        ------------------------------------------------------------
        -- OVERALL STATUS
        ------------------------------------------------------------

        CASE
            WHEN COALESCE(i.issue_record_count, 0) > 0
            THEN 'ISSUE DETECTED'
            ELSE 'HEALTHY'
        END AS monitoring_status,

        ------------------------------------------------------------
        -- PIPELINE SUMMARY
        ------------------------------------------------------------

        COALESCE(h.total_tables, 0)
            AS total_tables,

        COALESCE(h.successful_tables, 0)
            AS successful_tables,

        COALESCE(i.affected_table_count, 0)
            AS affected_table_count,

        ------------------------------------------------------------
        -- ISSUE COUNTS
        ------------------------------------------------------------

        COALESCE(i.failed_table_count, 0)
            AS failed_table_count,

        COALESCE(i.skipped_table_count, 0)
            AS skipped_table_count,

        COALESCE(i.unknown_status_count, 0)
            AS unknown_status_count,

        COALESCE(i.row_count_mismatch_count, 0)
            AS row_count_mismatch_count,

        COALESCE(i.schema_change_count, 0)
            AS schema_change_count,

        ------------------------------------------------------------
        -- TABLE LISTS
        ------------------------------------------------------------

        i.failed_tables,
        i.skipped_tables,
        i.unknown_status_tables,
        i.row_count_mismatch_tables,
        i.schema_change_tables,

        ------------------------------------------------------------
        -- DASHBOARD
        ------------------------------------------------------------

        'https://adb-3157519905744188.8.azuredatabricks.net/dashboardsv3/01f1a173c1941a8781307551adc9950e/published?o=3157519905744188'
            AS monitoring_dashboard_url

    FROM latest_run l

    LEFT JOIN health h
        ON h.run_date = l.run_date

    CROSS JOIN issue_rollup i

)

SELECT

    *,

    ------------------------------------------------------------
    -- FINAL TEAMS ALERT MESSAGE
    --
    -- CONCAT_WS skips NULL values, so issue types with a count
    -- of zero will NOT appear in Teams.
    ------------------------------------------------------------

    CASE

        WHEN alert_trigger = 1

        THEN CONCAT_WS(

            CHAR(10),

            CONCAT(
                'Run Date: ',
                CAST(run_date AS STRING)
            ),

            CONCAT(
                'Tables Checked: ',
                CAST(total_tables AS STRING),
                ' | Successful: ',
                CAST(successful_tables AS STRING),
                ' | Affected: ',
                CAST(affected_table_count AS STRING)
            ),

            '',

            --------------------------------------------------------
            -- FAILED TABLES
            --------------------------------------------------------

            CASE
                WHEN failed_table_count > 0
                THEN CONCAT(
                    '🔴 Failed Tables (',
                    CAST(failed_table_count AS STRING),
                    '): ',
                    failed_tables
                )
            END,

            --------------------------------------------------------
            -- SKIPPED TABLES
            --------------------------------------------------------

            CASE
                WHEN skipped_table_count > 0
                THEN CONCAT(
                    '🔴 Skipped Tables (',
                    CAST(skipped_table_count AS STRING),
                    '): ',
                    skipped_tables
                )
            END,

            --------------------------------------------------------
            -- ROW-COUNT MISMATCH
            --------------------------------------------------------

            CASE
                WHEN row_count_mismatch_count > 0
                THEN CONCAT(
                    '🟠 Row-Count Mismatch (',
                    CAST(row_count_mismatch_count AS STRING),
                    '): ',
                    row_count_mismatch_tables
                )
            END,

            --------------------------------------------------------
            -- UNKNOWN STATUS
            --------------------------------------------------------

            CASE
                WHEN unknown_status_count > 0
                THEN CONCAT(
                    '🟠 Unknown Status (',
                    CAST(unknown_status_count AS STRING),
                    '): ',
                    unknown_status_tables
                )
            END,

            --------------------------------------------------------
            -- SCHEMA CHANGE
            --------------------------------------------------------

            CASE
                WHEN schema_change_count > 0
                THEN CONCAT(
                    '🟡 Schema Changes (',
                    CAST(schema_change_count AS STRING),
                    '): ',
                    schema_change_tables
                )
            END,

            '',

            CONCAT(
                'Dashboard: ',
                monitoring_dashboard_url
            )

        )

        ELSE NULL

    END AS alert_message

FROM base;
