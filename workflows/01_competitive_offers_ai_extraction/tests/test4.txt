SELECT
  Quarter, QGP_Week, Week_Beginning_Monday, Week_Ending_Sunday, week_type,
  LOB_Supported,
  Spend_Actual, Spend_Forecast, Spend_Final
FROM prdrzranalytics.lab42.sdi_vw_mfc_gold_spend_weekly
WHERE Quarter IN ('2026 Q2', '2026 Q3')
ORDER BY QGP_Week DESC, LOB_Supported;

SELECT
  Quarter, QGP_Week, Week_Beginning_Monday, Week_Ending_Sunday, week_type,
  LOB_Supported, Channel, Tactic, Message_Type, Agency,
  Spend_Actual, Spend_Forecast, Spend_Final
FROM prdrzranalytics.lab42.sdi_vw_mfc_gold_spendGranular_weekly
WHERE Quarter IN ('2026 Q2', '2026 Q3')
ORDER BY QGP_Week DESC, LOB_Supported, Channel, Tactic, Message_Type, Agency;

SELECT
  Quarter, QGP_Week, Week_Beginning_Monday, Week_Ending_Sunday, week_type,
  LOB_Supported,
  SUM(Spend_Actual)   AS Spend_Actual,
  SUM(Spend_Forecast) AS Spend_Forecast,
  SUM(Spend_Final)    AS Spend_Final
FROM prdrzranalytics.lab42.sdi_vw_mfc_gold_spendGranular_weekly
WHERE Quarter IN ('2026 Q2', '2026 Q3')
GROUP BY Quarter, QGP_Week, Week_Beginning_Monday, Week_Ending_Sunday, week_type, LOB_Supported
ORDER BY QGP_Week DESC, LOB_Supported;

SELECT
  ng.QGP_Week, ng.LOB_Supported,
  ng.Spend_Actual AS NonGranular_Actual, g.Spend_Actual AS Granular_Actual,
  ng.Spend_Forecast AS NonGranular_Forecast, g.Spend_Forecast AS Granular_Forecast
FROM (
  SELECT QGP_Week, LOB_Supported, Spend_Actual, Spend_Forecast
  FROM prdrzranalytics.lab42.sdi_vw_mfc_gold_spend_weekly
  WHERE Quarter IN ('2026 Q2', '2026 Q3')
) ng
JOIN (
  SELECT QGP_Week, LOB_Supported, SUM(Spend_Actual) AS Spend_Actual, SUM(Spend_Forecast) AS Spend_Forecast
  FROM prdrzranalytics.lab42.sdi_vw_mfc_gold_spendGranular_weekly
  WHERE Quarter IN ('2026 Q2', '2026 Q3')
  GROUP BY QGP_Week, LOB_Supported
) g
  ON ng.QGP_Week = g.QGP_Week AND ng.LOB_Supported = g.LOB_Supported
WHERE ABS(COALESCE(ng.Spend_Actual, 0) - COALESCE(g.Spend_Actual, 0)) > 0.01
   OR ABS(COALESCE(ng.Spend_Forecast, 0) - COALESCE(g.Spend_Forecast, 0)) > 0.01
ORDER BY ng.QGP_Week DESC;
