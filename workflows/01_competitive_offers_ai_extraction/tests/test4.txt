WITH Deduped AS (
  SELECT
    *,
    ROW_NUMBER() OVER (
      PARTITION BY WeekEnding, MetricID, DateContext, MetricType, Page
      ORDER BY InsertDateTime DESC, PublishKey DESC
    ) AS rn
  FROM prdrzrlakehouse.qgp_restricted.qgpweeklyview
  WHERE DateContext = 'Normal'   -- confirmed true weekly grain; QTD/QTR are rollups of this
    AND MetricID IN ('VRInboundCallsinclHSIAutomatedManual', 'VRChatsinclHSIAutomatedManual')
)
SELECT
  WeekEnding,
  CASE WHEN MetricID = 'VRInboundCallsinclHSIAutomatedManual' THEN 'vrCalls' ELSE 'vrChats' END AS metric_name,
  CASE WHEN MetricType = 'QGP' THEN 'TARGET' ELSE 'ACTUAL' END AS side,
  Page,
  Amount
FROM Deduped
WHERE rn = 1
  AND (
    (MetricType != 'QGP' AND Page = 'Hero - Core Postpaid KP2')   -- actuals side
    OR
    (MetricType = 'QGP'  AND Page = 'Virtual Retail')              -- target side
  )
ORDER BY WeekEnding DESC, metric_name, side;
