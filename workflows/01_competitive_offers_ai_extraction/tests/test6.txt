-- PAID SOCIAL: prior completed week completeness by platform

WITH dates AS (
    SELECT
        date_sub(current_date(), dayofweek(current_date())) AS prior_week_end
),

base AS (
    SELECT
        REPLACE(TRIM(Channel_Name), 'Paid Social - ', '') AS platform,
        CAST(Date AS DATE) AS dt,
        TRY_CAST(Spend AS DOUBLE) AS spend
    FROM prdrzranalytics.lab42.media_analytics_integrated_snapshot
    WHERE Channel_Group_Name = 'Paid Social'
      AND Agency = 'InHouse'
      AND UPPER(TRIM(LOB)) IN ('POSTPAID', 'BROADBAND', 'HSI')
)

SELECT
    platform,
    COUNT(DISTINCT dt) AS days_present,
    MIN(dt) AS first_date,
    MAX(dt) AS latest_date,
    SUM(spend) AS spend,
    CASE
        WHEN COUNT(DISTINCT dt) = 7 THEN 'COMPLETE'
        ELSE 'INCOMPLETE'
    END AS status
FROM base
CROSS JOIN dates
WHERE dt BETWEEN date_sub(prior_week_end, 6) AND prior_week_end
GROUP BY platform
ORDER BY status DESC, platform;
