WITH
actual_boundary_totals AS (
  SELECT
    Week_Beginning_Monday, Week_Ending_Sunday, QGP_Week, LOB_Supported,
    SUM(weekly_actual) AS total_week_actual,
    MAX(FileLoad_Date) AS FileLoad_Date,
    MAX(Source_File_Date) AS Source_File_Date
  FROM prdrzranalytics.lab42.vw_mfc_bronze_spendActuals_weekly
  WHERE week_type = 'boundary_week'
  GROUP BY Week_Beginning_Monday, Week_Ending_Sunday, QGP_Week, LOB_Supported
),
actual_boundary_daily AS (
  SELECT
    t.Week_Beginning_Monday, t.Week_Ending_Sunday, t.QGP_Week, t.LOB_Supported,
    t.FileLoad_Date, t.Source_File_Date,
    day_date,
    t.total_week_actual / 7.0 AS daily_actual
  FROM actual_boundary_totals t
  LATERAL VIEW explode(sequence(t.Week_Beginning_Monday, t.Week_Ending_Sunday)) exploded AS day_date
),
actual_boundary_reallocated AS (
  SELECT
    CONCAT('Q', CAST(QUARTER(day_date) AS STRING), '''', LPAD(CAST(YEAR(day_date) % 100 AS STRING), 2, '0')) AS Quarter,
    Week_Beginning_Monday, Week_Ending_Sunday, QGP_Week, LOB_Supported,
    MAX(FileLoad_Date) AS FileLoad_Date,
    MAX(Source_File_Date) AS Source_File_Date,
    SUM(daily_actual) AS weekly_actual,
    'boundary_week' AS week_type
  FROM actual_boundary_daily
  GROUP BY QUARTER(day_date), YEAR(day_date), Week_Beginning_Monday, Week_Ending_Sunday, QGP_Week, LOB_Supported
)
SELECT Quarter, Week_Beginning_Monday, Week_Ending_Sunday, QGP_Week, LOB_Supported, FileLoad_Date, Source_File_Date, weekly_actual, week_type
FROM prdrzranalytics.lab42.vw_mfc_bronze_spendActuals_weekly
WHERE week_type != 'boundary_week'
UNION ALL
SELECT Quarter, Week_Beginning_Monday, Week_Ending_Sunday, QGP_Week, LOB_Supported, FileLoad_Date, Source_File_Date, weekly_actual, week_type
FROM actual_boundary_reallocated;
