-- ============================================================
-- UNIQUE CATEGORIES FOR ALL DIMENSION COLUMNS
--
-- Dimension definition:
--   STRING + BOOLEAN columns
--
-- Scope:
--   Programmatic = full source universe
--   Paid Social  = Channel_Group_Name = 'Paid Social'
--                  (all agencies / LOBs retained)
--   Paid Search  = full source universe
--
-- Output is long-form and Excel-friendly.
-- ============================================================

DECLARE OR REPLACE VARIABLE category_sql STRING;

SET VAR category_sql = (

    WITH source_columns AS (

        -- ====================================================
        -- 1. PROGRAMMATIC
        -- ====================================================
        SELECT
            1 AS source_order,
            'Programmatic' AS source_channel,
            'pbi_programmatic_browsers_currentyr' AS source_table,
            'prd_dbi_analytics.improvado.pbi_programmatic_browsers_currentyr'
                AS full_table_name,
            '1 = 1' AS row_filter,

            ordinal_position,
            column_name,
            data_type

        FROM prd_dbi_analytics.information_schema.columns

        WHERE table_schema = 'improvado'
          AND table_name = 'pbi_programmatic_browsers_currentyr'
          AND UPPER(data_type) IN ('STRING', 'BOOLEAN')


        UNION ALL


        -- ====================================================
        -- 2. PAID SOCIAL
        -- ====================================================
        SELECT
            2 AS source_order,
            'Paid Social' AS source_channel,
            'media_analytics_integrated_snapshot' AS source_table,
            'prdrzranalytics.lab42.media_analytics_integrated_snapshot'
                AS full_table_name,

            -- Universe for Paid Social only.
            -- Do NOT restrict Agency or LOB here.
            'Channel_Group_Name = ''Paid Social''' AS row_filter,

            ordinal_position,
            column_name,
            data_type

        FROM prdrzranalytics.information_schema.columns

        WHERE table_schema = 'lab42'
          AND table_name = 'media_analytics_integrated_snapshot'
          AND UPPER(data_type) IN ('STRING', 'BOOLEAN')


        UNION ALL


        -- ====================================================
        -- 3. PAID SEARCH
        -- ====================================================
        SELECT
            3 AS source_order,
            'Paid Search' AS source_channel,
            'sdi_tbl_sa360_gold_campaign_daily' AS source_table,
            'prdrzranalytics.lab42.sdi_tbl_sa360_gold_campaign_daily'
                AS full_table_name,
            '1 = 1' AS row_filter,

            ordinal_position,
            column_name,
            data_type

        FROM prdrzranalytics.information_schema.columns

        WHERE table_schema = 'lab42'
          AND table_name = 'sdi_tbl_sa360_gold_campaign_daily'
          AND UPPER(data_type) IN ('STRING', 'BOOLEAN')
    ),


    -- Build the STACK arguments dynamically.
    --
    -- Each dimension contributes:
    --   ordinal_position,
    --   dimension_name,
    --   data_type,
    --   value
    dimension_expressions AS (

        SELECT
            source_order,
            source_channel,
            source_table,
            full_table_name,
            row_filter,
            ordinal_position,

            CONCAT(
                CAST(ordinal_position AS STRING),
                ', ',
                '''',
                REPLACE(column_name, '''', ''''''),
                ''', ',
                '''',
                REPLACE(data_type, '''', ''''''),
                ''', ',
                'CAST(`',
                REPLACE(column_name, '`', '``'),
                '` AS STRING)'
            ) AS stack_expression

        FROM source_columns
    ),


    -- Build one query per source table.
    source_queries AS (

        SELECT
            source_order,

            CONCAT(
                'SELECT ',
                    '''', source_channel, ''' AS source_channel, ',
                    '''', source_table, ''' AS source_table, ',
                    'dimension_order, ',
                    'dimension_name, ',
                    'data_type, ',
                    'COALESCE(category_value, ''[NULL]'') AS category_value, ',
                    'COUNT(*) AS row_count ',

                'FROM ',
                    full_table_name, ' ',

                'LATERAL VIEW STACK(',
                    CAST(COUNT(*) AS STRING),
                    ', ',

                    ARRAY_JOIN(
                        TRANSFORM(
                            SORT_ARRAY(
                                COLLECT_LIST(
                                    NAMED_STRUCT(
                                        'ord', ordinal_position,
                                        'expr', stack_expression
                                    )
                                )
                            ),
                            x -> x.expr
                        ),
                        ', '
                    ),

                ') category_stack ',
                    'AS dimension_order, dimension_name, data_type, category_value ',

                'WHERE ',
                    row_filter, ' ',

                'GROUP BY ',
                    'dimension_order, ',
                    'dimension_name, ',
                    'data_type, ',
                    'category_value'
            ) AS source_sql

        FROM dimension_expressions

        GROUP BY
            source_order,
            source_channel,
            source_table,
            full_table_name,
            row_filter
    )


    -- Combine the three generated queries.
    SELECT
        CONCAT(
            'SELECT * FROM (',

            ARRAY_JOIN(
                TRANSFORM(
                    SORT_ARRAY(
                        COLLECT_LIST(
                            NAMED_STRUCT(
                                'ord', source_order,
                                'sql', source_sql
                            )
                        )
                    ),
                    x -> x.sql
                ),
                ' UNION ALL '
            ),

            ') categories ',
            'ORDER BY ',
                'source_channel, ',
                'dimension_order, ',
                'category_value'
        )

    FROM source_queries
);


-- Run dynamically generated query.
EXECUTE IMMEDIATE category_sql;

WITH SourceWeekly AS (

  SELECT
    date_add(CAST(date AS DATE), 7 - dayofweek(CAST(date AS DATE))) AS week_ending,

    CASE
      WHEN UPPER(TRIM(lob)) = 'POSTPAID'
        THEN 'POSTPAID'

      WHEN UPPER(TRIM(lob)) IN ('HSI', 'BROADBAND')
        THEN 'BROADBAND'

      WHEN UPPER(TRIM(lob)) = 'FIBER'
        THEN 'FIBER'
    END AS lob,

    SUM(cost) AS source_spend

  FROM prdrzranalytics.lab42.sdi_tbl_sa360_gold_campaign_daily

  WHERE UPPER(TRIM(ad_platform)) IN (
    'GOOGLE',
    'GOOGLE ADS',
    'GOOGLE_ADS',
    'BING',
    'MICROSOFT',
    'MICROSOFT ADS',
    'MICROSOFT_ADS'
  )

    AND UPPER(TRIM(lob)) IN (
      'POSTPAID',
      'HSI',
      'BROADBAND',
      'FIBER'
    )

  GROUP BY
    date_add(CAST(date AS DATE), 7 - dayofweek(CAST(date AS DATE))),

    CASE
      WHEN UPPER(TRIM(lob)) = 'POSTPAID'
        THEN 'POSTPAID'

      WHEN UPPER(TRIM(lob)) IN ('HSI', 'BROADBAND')
        THEN 'BROADBAND'

      WHEN UPPER(TRIM(lob)) = 'FIBER'
        THEN 'FIBER'
    END
),


SourcePivot AS (

  SELECT
    week_ending,

    SUM(
      CASE
        WHEN lob = 'POSTPAID'
        THEN source_spend
      END
    ) AS source_postpaid,

    SUM(
      CASE
        WHEN lob = 'BROADBAND'
        THEN source_spend
      END
    ) AS source_broadband,

    SUM(
      CASE
        WHEN lob = 'FIBER'
        THEN source_spend
      END
    ) AS source_fiber,

    SUM(source_spend) AS source_total

  FROM SourceWeekly

  GROUP BY week_ending
),


SilverPivot AS (

  SELECT
    qgp_date,

    MAX(
      CASE
        WHEN lob = 'POSTPAID'
        THEN metric_value
      END
    ) AS silver_postpaid,

    MAX(
      CASE
        WHEN lob = 'BROADBAND'
        THEN metric_value
      END
    ) AS silver_broadband,

    MAX(
      CASE
        WHEN lob = 'FIBER'
        THEN metric_value
      END
    ) AS silver_fiber,

    MAX(
      CASE
        WHEN lob = 'ALL'
        THEN metric_value
      END
    ) AS silver_total

  FROM prdrzranalytics.lab42.sdi_tbl_dashboardPulseTms_silver_biddableSpend_weekly

  WHERE data_source = 'BIDDABLE_SPEND_CHANNEL'
    AND metric_name = 'biddableSpend'
    AND channel_group = 'Paid Search - All'
    AND lob IN (
      'POSTPAID',
      'BROADBAND',
      'FIBER',
      'ALL'
    )

    /* Clean source reconciliation:
       excludes QGP boundary-prorated rows */
    AND week_type = 'NORMAL'

  GROUP BY qgp_date
)


SELECT
  s.qgp_date,

  /* SOURCE */
  src.source_postpaid,
  src.source_broadband,
  src.source_fiber,
  src.source_total,

  /* SILVER */
  s.silver_postpaid,
  s.silver_broadband,
  s.silver_fiber,
  s.silver_total,

  /* RECONCILIATION */
  s.silver_postpaid - src.source_postpaid AS postpaid_diff,
  s.silver_broadband - src.source_broadband AS broadband_diff,
  s.silver_fiber - src.source_fiber AS fiber_diff,
  s.silver_total - src.source_total AS total_diff,

  /* SILVER INTERNAL QA */
  s.silver_total
    - (
        COALESCE(s.silver_postpaid, 0)
        + COALESCE(s.silver_broadband, 0)
        + COALESCE(s.silver_fiber, 0)
      ) AS silver_total_component_diff

FROM SilverPivot s

LEFT JOIN SourcePivot src
  ON src.week_ending = s.qgp_date

ORDER BY
  s.qgp_date DESC;
