// c.Bidd Channel Match

IF [p.tms.Bidd Channel] = 'All Channels' THEN

    [Channel Group] IN (
        'Paid Search',
        'Paid Social',
        'Programmatic'
    )

ELSEIF [p.tms.Bidd Channel] = 'Paid Search - All' THEN

    [Channel Group] = 'Paid Search'

ELSEIF [p.tms.Bidd Channel] = 'Paid Social - All' THEN

    [Channel Group] = 'Paid Social'

ELSEIF [p.tms.Bidd Channel] = 'Programmatic - All' THEN

    [Channel Group] = 'Programmatic'

ELSEIF STARTSWITH([p.tms.Bidd Channel], 'Paid Search - ') THEN

    [Channel Group] = 'Paid Search'
    AND [Platform] =
        REPLACE([p.tms.Bidd Channel], 'Paid Search - ', '')

ELSEIF STARTSWITH([p.tms.Bidd Channel], 'Paid Social - ') THEN

    [Channel Group] = 'Paid Social'
    AND [Platform] =
        REPLACE([p.tms.Bidd Channel], 'Paid Social - ', '')

ELSEIF STARTSWITH([p.tms.Bidd Channel], 'Programmatic - ') THEN

    [Channel Group] = 'Programmatic'
    AND [Platform] =
        REPLACE([p.tms.Bidd Channel], 'Programmatic - ', '')

ELSE
    FALSE
END
