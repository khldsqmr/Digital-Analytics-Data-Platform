SELECT data_source, COUNT(*) AS row_count, MAX(week_sun_to_sat) AS most_recent_week, MIN(week_sun_to_sat) AS oldest_week
FROM prdrzranalytics.lab42.sdi_tbl_dashboardPulseByod_gold_unified_long
GROUP BY data_source
ORDER BY data_source;

SELECT week_sun_to_sat, metric_value, metric_value_wow, wow_pct
FROM prdrzranalytics.lab42.sdi_tbl_dashboardPulseByod_gold_unified_long
WHERE data_source = 'ADOBE' AND metric_name = 'adobe_ordersAssistedByod_social'
ORDER BY week_sun_to_sat DESC
LIMIT 15;
