-- PROGRAMMATIC: source universe + what our Biddable scope keeps

SELECT
    DSP,
    account_id,
    lob,
    media_partner,
    Buy_Type,
    Channel,

    CASE
        WHEN UPPER(TRIM(lob)) IN ('POSTPAID', 'HSI', 'BROADBAND')
            THEN 'KEEP - PP/BB'
        ELSE 'EXCLUDE - OTHER LOB'
    END AS biddable_scope,

    COUNT(*) AS rows,
    SUM(TRY_CAST(spend AS DOUBLE)) AS spend,
    MIN(date) AS min_date,
    MAX(date) AS max_date

FROM prd_dbi_analytics.improvado.pbi_programmatic_browsers_currentyr

WHERE date >= date_sub(current_date(), 35)

GROUP BY
    DSP,
    account_id,
    lob,
    media_partner,
    Buy_Type,
    Channel,
    CASE
        WHEN UPPER(TRIM(lob)) IN ('POSTPAID', 'HSI', 'BROADBAND')
            THEN 'KEEP - PP/BB'
        ELSE 'EXCLUDE - OTHER LOB'
    END

ORDER BY DSP, spend DESC;

-- PAID SOCIAL: source universe + our intended reporting scope

SELECT
    Agency,
    Account_ID,
    Account_Name,
    Account_Type,
    LOB,
    Channel_Name,

    CASE
        WHEN UPPER(TRIM(Agency)) = 'INHOUSE'
         AND UPPER(TRIM(LOB)) IN ('POSTPAID', 'BROADBAND', 'HSI')
            THEN 'KEEP - INHOUSE PP/BB'

        WHEN UPPER(TRIM(Agency)) <> 'INHOUSE'
            THEN 'EXCLUDE - NOT INHOUSE'

        ELSE 'EXCLUDE - OTHER LOB'
    END AS biddable_scope,

    COUNT(*) AS rows,
    SUM(TRY_CAST(Spend AS DOUBLE)) AS spend,
    MIN(CAST(Date AS DATE)) AS min_date,
    MAX(CAST(Date AS DATE)) AS max_date

FROM prdrzranalytics.lab42.media_analytics_integrated_snapshot

WHERE Channel_Group_Name = 'Paid Social'
  AND Date >= date_sub(current_date(), 35)

GROUP BY
    Agency,
    Account_ID,
    Account_Name,
    Account_Type,
    LOB,
    Channel_Name,
    CASE
        WHEN UPPER(TRIM(Agency)) = 'INHOUSE'
         AND UPPER(TRIM(LOB)) IN ('POSTPAID', 'BROADBAND', 'HSI')
            THEN 'KEEP - INHOUSE PP/BB'

        WHEN UPPER(TRIM(Agency)) <> 'INHOUSE'
            THEN 'EXCLUDE - NOT INHOUSE'

        ELSE 'EXCLUDE - OTHER LOB'
    END

ORDER BY Agency, spend DESC;
