-- 1. Raw QGP source — is fresh data even there yet?
SELECT MAX(InsertDateTime) AS latest_insert, MAX(WeekEnding) AS latest_week
FROM prdrzrlakehouse.qgp_restricted.qgpweeklyview
WHERE DateContext = 'Normal';

-- 2. Archive Bronze — did the archive's MERGE pick it up?
SELECT MAX(InsertDateTime) AS latest_insert, MAX(WeekEnding) AS latest_week
FROM prdrzranalytics.lab42.sdi_tbl_qgpArchive_bronze_retained_weekly;

-- 3. Archive Gold — matches what you already checked
SELECT MAX(WeekEnding) AS latest_week
FROM prdrzranalytics.lab42.sdi_tbl_qgpArchive_gold_curated_weekly
WHERE DateContext = 'Normal';

-- 4. PulseTMS's OWN QGP Bronze — reads FROM the archive; is it keeping up with it?
SELECT MAX(qgp_date) AS latest_date
FROM prdrzranalytics.lab42.sdi_tbl_dashboardPulseTms_bronze_qgp_weekly;

-- 5. The actual Gold output you'd see on a dashboard
SELECT MAX(qgp_date) AS latest_date,
       MAX(CASE WHEN metric_value IS NOT NULL THEN qgp_date END) AS latest_with_data
FROM prdrzranalytics.lab42.sdi_vw_dashboardPulseTms_gold_unified_long
WHERE data_source = 'QGP_SCORECARD';
