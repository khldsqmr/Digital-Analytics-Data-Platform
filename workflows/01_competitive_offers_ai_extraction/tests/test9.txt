CREATE TABLE sdi_tbl_mip_bronze_edlSnapshotTest_ref AS
SELECT row_identity_hash, load_datetime_pst, pipeline_batch_id, event_page_view, customer_id, channel_name
FROM prd_dbi_analytics.silver_digital_interactions.unified_digital_interactions
WHERE event_date = DATE '2026-09-26'
  AND pmod(abs(hash(row_identity_hash)), 100) = 0;

  WITH now_rows AS (
  SELECT row_identity_hash, load_datetime_pst, event_page_view, customer_id, channel_name
  FROM prd_dbi_analytics.silver_digital_interactions.unified_digital_interactions
  WHERE event_date = DATE '2026-09-26'
    AND pmod(abs(hash(row_identity_hash)), 100) = 0
)
SELECT
  SUM(CASE WHEN n.row_identity_hash IS NULL THEN 1 ELSE 0 END) AS deleted_or_rehashed,
  SUM(CASE WHEN o.row_identity_hash IS NULL THEN 1 ELSE 0 END) AS newly_arrived,
  SUM(CASE WHEN o.row_identity_hash IS NOT NULL AND n.row_identity_hash IS NOT NULL
            AND (NOT (o.load_datetime_pst <=> n.load_datetime_pst)
              OR NOT (o.customer_id     <=> n.customer_id)
              OR NOT (o.event_page_view <=> n.event_page_view)
              OR NOT (o.channel_name    <=> n.channel_name)) THEN 1 ELSE 0 END) AS updated_in_place
FROM sdi_tbl_mip_bronze_edlSnapshotTest_ref o
FULL OUTER JOIN now_rows n ON o.row_identity_hash = n.row_identity_hash;

SELECT
  u.source_table,
  datediff(to_date(u.load_datetime_pst), u.event_date) AS days_late,
  COUNT(*)                                             AS hits,
  SUM(CASE WHEN s.row_identity_hash IS NULL THEN 1 ELSE 0 END) AS hits_without_session

LEFT JOIN prd_dbi_analytics.silver_digital_interactions.session_event_fact s
  ON  s.row_identity_hash = u.row_identity_hash
  AND s.event_date        = u.event_date
  AND s.source_table      = u.source_table
WHERE u.event_date = DATE '2026-09-24'

ORDER BY 1, 2;

SELECT
  u.source_table,
  COUNT(*)                                                         AS hits,
  SUM(CASE WHEN s.canonical_user_id IS NULL THEN 1 ELSE 0 END)     AS null_canonical,
  SUM(CASE WHEN u.customer_id IS NOT NULL
            AND s.canonical_user_id = u.customer_id THEN 1 ELSE 0 END) AS canonical_equals_customer_id,
  COUNT(DISTINCT s.canonical_user_id)                              AS distinct_canonical,
  COUNT(DISTINCT s.session_id)                                     AS distinct_sessions
FROM prd_dbi_analytics.silver_digital_interactions.session_event_fact s
JOIN prd_dbi_analytics.silver_digital_interactions.unified_digital_interactions u
  ON  s.row_identity_hash = u.row_identity_hash
  AND s.event_date        = u.event_date
  AND s.source_table      = u.source_table
WHERE s.event_date = DATE '2026-09-24'
GROUP BY 1;

SELECT
  COUNT(*)                                                               AS sessions,
  SUM(CASE WHEN to_date(session_end_time) > session_start_date THEN 1 ELSE 0 END) AS cross_midnight,

FROM prd_dbi_analytics.silver_digital_interactions.session_summary_fact
WHERE session_start_date = DATE '2026-09-24';
