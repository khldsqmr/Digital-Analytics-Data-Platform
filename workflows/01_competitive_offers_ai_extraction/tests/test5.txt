  'CONSUMERPOSTPAIDNEWPHONEBANNONBOPISUNASSISTEDACTIVATIONSTM1MAPPEDDIGITAL',

  'NEWPHONEBANASSISTEDACTIVATIONSDIGITALPCTOFCONSUMERPOSTPAIDTM1MAPPEDWONRINDIRECT'
